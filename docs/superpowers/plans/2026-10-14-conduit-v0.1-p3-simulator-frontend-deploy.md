```java
import org.junit.jupiter.api.BeforeEach;

import java.util.concurrent.CopyOnWriteArrayList;
import java.util.concurrent.atomic.AtomicInteger;

import static org.assertj.core.api.Assertions.assertThat;

class DemoProvisionerTest {

    HttpServer server;
    final List<String> calls = new CopyOnWriteArrayList<>();
    final List<String> authHeaders = new CopyOnWriteArrayList<>();
    final AtomicInteger created = new AtomicInteger();

    @BeforeEach
    void start() throws IOException {
        server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);

        server.start();

    private void handle(HttpExchange ex) throws IOException {
        String call = ex.getRequestMethod() + " " + ex.getRequestURI().getPath();
        calls.add(call);
        if (!call.endsWith("/auth/login")) {
            authHeaders.add(ex.getRequestHeaders().getFirst("Authorization"));
        }
        String body;
        int status = 200;
        switch (call) {
            case "POST /api/v1/auth/login" -> body = "{\"accessToken\":\"tok\"}";
            case "GET /api/v1/products" -> body = """
                    {"items":[{"productKey":"pk-demo","name":"Demo Meter"}],"total":1,"page":1,"size":100}""";

            default -> {
                status = 404;
                body = "{}";
            }
        }
        byte[] bytes = body.getBytes(StandardCharsets.UTF_8);
        ex.getResponseHeaders().add("Content-Type", "application/json");
        ex.sendResponseHeaders(status, bytes.length);
        ex.getResponseBody().write(bytes);
```

```java
package dev.conduit.simulator;

import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Random;

/** 生成三相电表读数：电压围绕 220V、电流围绕 5A 波动，电能按功率累加。 */
public class ReadingGenerator {

    private final Random random;
    private double energyKwh;

    public ReadingGenerator(long seed) {
        this.random = new Random(seed);
        this.energyKwh = 1000 + random.nextInt(9000);
    }

    public Map<String, Object> next(double intervalSeconds) {
        Map<String, Object> reading = new LinkedHashMap<>();
        double powerW = 0;
        for (String phase : new String[]{"a", "b", "c"}) {
            double voltage = round(clamp(220 + random.nextGaussian() * 1.5, 211, 229), 1);
            double current = round(clamp(5 + random.nextGaussian() * 0.8, 0.5, 9.5), 2);
            reading.put("U" + phase, voltage);
            reading.put("I" + phase, current);
            powerW += voltage * current;
        }
        // 保证严格递增：即使功率极小也至少增加 0.001 kWh
        energyKwh += Math.max(powerW / 1000 * intervalSeconds / 3600, 0.001);
        reading.put("EPimp", round(energyKwh, 3));
        return reading;
    }

    private static double clamp(double v, double min, double max) {
        return Math.max(min, Math.min(max, v));
    }

    private static double round(double v, int digits) {
        double scale = Math.pow(10, digits);
        return Math.round(v * scale) / scale;
    }
}
```

```java
package dev.conduit.simulator;

import com.fasterxml.jackson.core.JsonProcessingException;

import java.io.UncheckedIOException;
import java.util.LinkedHashMap;
import java.util.Map;

public final class PayloadFactory {

    }

    public static byte[] property(String id, long ts, Map<String, Object> params) {
        Map<String, Object> body = new LinkedHashMap<>();
        body.put("id", id);
        body.put("ts", ts);
        body.put("params", params);
        try {
            return MAPPER.writeValueAsBytes(body);
        } catch (JsonProcessingException e) {
            throw new UncheckedIOException(e);
        }
    }
}
```

```java
package dev.conduit.simulator;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.io.IOException;
import java.io.UncheckedIOException;
import java.net.URI;
import java.net.URLEncoder;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;

/** Conduit REST 的最小客户端，只覆盖模拟器与端到端测试需要的接口。 */
public class ConduitApiClient {

    private static final int PAGE_SIZE = 100;

    private final URI baseUrl;
    private final HttpClient http = HttpClient.newBuilder().connectTimeout(Duration.ofSeconds(5)).build();
    private final ObjectMapper mapper = new ObjectMapper();
    private String token;

    public ConduitApiClient(URI baseUrl) {
        this.baseUrl = baseUrl;
    }

    public void login(String username, String password) {
        JsonNode body = send("POST", "/api/v1/auth/login", Map.of("username", username, "password", password), 200);
        token = body.get("accessToken").asText();
    }

    public String findOrCreateProduct(String name) {
        for (int page = 1; ; page++) {
            JsonNode result = send("GET", "/api/v1/products?page=" + page + "&size=" + PAGE_SIZE, null, 200);
            for (JsonNode item : result.get("items")) {
                if (name.equals(item.get("name").asText())) {
                    return item.get("productKey").asText();
                }
            }
            if ((long) page * PAGE_SIZE >= result.get("total").asLong()) {
                break;
            }
        }
        return send("POST", "/api/v1/products", Map.of("name", name), 201).get("productKey").asText();
    }

    public List<String> listDeviceKeys(String productKey) {
        List<String> keys = new ArrayList<>();
        for (int page = 1; ; page++) {
            JsonNode result = send("GET", "/api/v1/devices?productKey=" + encode(productKey)
                    + "&page=" + page + "&size=" + PAGE_SIZE, null, 200);
            result.get("items").forEach(item -> keys.add(item.get("deviceKey").asText()));
            if ((long) page * PAGE_SIZE >= result.get("total").asLong()) {
                return keys;
            }
        }
    }

    public DeviceCredentials createDevice(String productKey, String name) {
        JsonNode body = send("POST", "/api/v1/devices", Map.of("productKey", productKey, "name", name), 201);
        return new DeviceCredentials(productKey, body.get("device").get("deviceKey").asText(),
                body.get("deviceSecret").asText());
    }

    public DeviceCredentials resetSecret(String productKey, String deviceKey) {
        JsonNode body = send("POST", "/api/v1/devices/" + encode(deviceKey) + "/reset-secret", null, 200);
        return new DeviceCredentials(productKey, deviceKey, body.get("deviceSecret").asText());
    }

    private JsonNode send(String method, String path, Object body, int expectedStatus) {
        try {
            HttpRequest.Builder request = HttpRequest.newBuilder(baseUrl.resolve(path))
                    .timeout(Duration.ofSeconds(10))
                    .header("Content-Type", "application/json");
            if (token != null) {
                request.header("Authorization", "Bearer " + token);
            }
            request.method(method, body == null
                    ? HttpRequest.BodyPublishers.noBody()
                    : HttpRequest.BodyPublishers.ofByteArray(mapper.writeValueAsBytes(body)));

            HttpResponse<byte[]> response = http.send(request.build(), HttpResponse.BodyHandlers.ofByteArray());
            if (response.statusCode() != expectedStatus) {
                throw new IllegalStateException(method + " " + path + " returned " + response.statusCode()
                        + ": " + new String(response.body(), StandardCharsets.UTF_8));
            }
            return mapper.readTree(response.body());
        } catch (IOException e) {
            throw new UncheckedIOException(e);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new IllegalStateException(e);
        }
    }

    private static String encode(String value) {
        return URLEncoder.encode(value, StandardCharsets.UTF_8);
    }
}
```

```java
package dev.conduit.simulator;

import com.hivemq.client.mqtt.MqttClient;
import com.hivemq.client.mqtt.datatypes.MqttQos;
import com.hivemq.client.mqtt.mqtt5.Mqtt5AsyncClient;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.Executors;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;

/** 每台设备一个 MQTT 连接，按固定周期以 QoS1 上报；各设备的首次上报在一个周期内均匀错开。 */
public class DeviceSimulator {

    private static final Logger log = LoggerFactory.getLogger(DeviceSimulator.class);

    private final String host;
    private final int port;
    private final Duration interval;
    private final int messagesPerDevice;

    public DeviceSimulator(String host, int port, Duration interval, int messagesPerDevice) {
        this.host = host;
        this.port = port;
        this.interval = interval;
        this.messagesPerDevice = messagesPerDevice;
    }

    public void run(List<DeviceCredentials> devices) throws InterruptedException {
        ScheduledExecutorService scheduler = Executors.newScheduledThreadPool(4);
        CountDownLatch finished = new CountDownLatch(messagesPerDevice > 0 ? devices.size() : 1);
        List<Mqtt5AsyncClient> clients = new ArrayList<>();
        long intervalMs = interval.toMillis();

        for (int i = 0; i < devices.size(); i++) {
            DeviceCredentials d = devices.get(i);
            Mqtt5AsyncClient client = MqttClient.builder().useMqttVersion5()
                    .identifier(d.deviceKey())
                    .serverHost(host).serverPort(port)
                    .automaticReconnectWithDefaultConfig()
                    .simpleAuth().username(d.deviceKey()).password(d.secret().getBytes(StandardCharsets.UTF_8))
                    .applySimpleAuth()
                    .buildAsync();
            client.connect().join();
            clients.add(client);

            ReadingGenerator generator = new ReadingGenerator(d.deviceKey().hashCode());
            String topic = "conduit/%s/%s/up/property".formatted(d.productKey(), d.deviceKey());
            AtomicInteger sent = new AtomicInteger();
            AtomicInteger acked = new AtomicInteger();
            long initialDelay = intervalMs * i / Math.max(1, devices.size());

            scheduler.scheduleAtFixedRate(() -> {
                if (messagesPerDevice > 0 && sent.get() >= messagesPerDevice) {
                    return;
                }
                int seq = sent.incrementAndGet();
                byte[] payload = PayloadFactory.property(d.deviceKey() + "-" + seq, System.currentTimeMillis(),
                        generator.next(intervalMs / 1000.0));
                client.publishWith().topic(topic).qos(MqttQos.AT_LEAST_ONCE).payload(payload).send()
                        .whenComplete((ack, error) -> {
                            if (error != null) {
                                log.warn("Publish failed for {}: {}", d.deviceKey(), error.getMessage());
                            } else if (messagesPerDevice > 0 && acked.incrementAndGet() == messagesPerDevice) {
                                finished.countDown();
                            }
                        });
            }, initialDelay, intervalMs, TimeUnit.MILLISECONDS);
        }

        log.info("Simulating {} devices every {} ms", devices.size(), intervalMs);
        finished.await();
        scheduler.shutdownNow();
        clients.forEach(c -> c.disconnect().join());
    }
}
```

```ts
import axios, { type AxiosError, type AxiosInstance } from 'axios'

declare module 'axios' {
  interface AxiosRequestConfig {
    /** 已经因 401 重试过一次，不再刷新 */
    _retried?: boolean
  }
}

export interface TokenHolder {
  getToken(): string | null
  setToken(token: string | null): void
  onAuthFailed(): void
}

export const REFRESH_URL = '/api/v1/auth/refresh'
const AUTH_PREFIX = '/api/v1/auth/'

/**
 * 创建带鉴权的 Axios 实例：自动附带 access token；遇到 401 时用 refresh Cookie 换新 token 并重试一次，
 * 并发的多个 401 共享同一次刷新。刷新失败则清空 token 并通知调用方（跳转登录）。
 */
export function createHttp(holder: TokenHolder): AxiosInstance {
  const http = axios.create({ withCredentials: true })
  let refreshing: Promise<string> | null = null

  function refreshToken(): Promise<string> {
    refreshing ??= http
      .post<{ accessToken: string }>(REFRESH_URL, null, { _retried: true })
      .then((res) => {
        holder.setToken(res.data.accessToken)
        return res.data.accessToken
      })
      .finally(() => {
        refreshing = null
      })
    return refreshing
  }

  http.interceptors.request.use((config) => {
    const token = holder.getToken()
    if (token && !config.url?.startsWith(AUTH_PREFIX)) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  })

  http.interceptors.response.use(undefined, async (error: AxiosError) => {
    const config = error.config
    if (error.response?.status !== 401 || !config || config._retried || config.url?.startsWith(AUTH_PREFIX)) {
      return Promise.reject(error)
    }
    config._retried = true
    try {
      const token = await refreshToken()
      config.headers.Authorization = `Bearer ${token}`
      return http(config)
    } catch (refreshError) {
      holder.setToken(null)
      holder.onAuthFailed()
      return Promise.reject(refreshError)
    }
  })

  return http
}
```

```ts
/** 从接口错误中提取给用户看的文案：优先 ProblemDetail.detail。 */
export function problemMessage(error: unknown): string {
  if (isAxiosError(error)) {
    const detail = (error.response?.data as { detail?: unknown } | undefined)?.detail
    if (typeof detail === 'string' && detail) {
      return detail
    }
    if (error.response) {
      return `请求失败（${error.response.status}）`
    }
    return '无法连接服务器'
  }
  return error instanceof Error ? error.message : '未知错误'
}
```

```ts
    token: null as string | null,

    async login(username: string, password: string) {
      const res = await authApi.login(username, password)
      this.token = res.accessToken
      this.user = res.user
    },

    async restore(): Promise<boolean> {

        this.user = res.user
        return true
      } catch {
        this.clear()
        return false
      }
    },
    async logout() {
      try {
        await authApi.logout()
      } finally {
        this.clear()
      }
    },
    clear() {
      this.token = null
      this.user = null
    },
  },
```

```ts
export function formatDateTime(value: string | number | null | undefined): string {
  if (value === null || value === undefined || value === '') {
    return '—'
  }
  const d = new Date(value)
  if (Number.isNaN(d.getTime())) {
    return '—'
  }
  const p = (n: number) => String(n).padStart(2, '0')
  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}:${p(d.getSeconds())}`
}
```

```ts
import { onBeforeUnmount, onMounted } from 'vue'

/** 立即执行一次，之后每次执行结束后等待 intervalMs 再执行；失败不影响后续轮询。 */
export function startPolling(task: () => unknown, intervalMs: number): () => void {
  let stopped = false
  let timer: ReturnType<typeof setTimeout> | undefined

  const tick = async () => {
    if (stopped) {
      return
    }
    try {
      await task()
    } catch {
      // 下一轮再试
    } finally {
      if (!stopped) {
        timer = setTimeout(tick, intervalMs)
      }
    }
  }

  void tick()
  return () => {
    stopped = true
    if (timer) {
      clearTimeout(timer)
    }
  }
}

export function usePolling(task: () => unknown, intervalMs: number) {
  let stop: () => void = () => {}
  onMounted(() => {
    stop = startPolling(task, intervalMs)
  })
  onBeforeUnmount(() => stop())
}
```

```vue
const createForm = reactive({ productKey: '', name: '' })

const secretVisible = ref(false)
const secret = reactive({ deviceKey: '', value: '' })

async function loadProducts() {
  try {
    products.value = (await productApi.list(1, 100)).items
  } catch (e) {
    ElMessage.error(problemMessage(e))
  }
}

async function load() {
  loading.value = true
  try {
    const res = await deviceApi.list({
      productKey: query.productKey || undefined,
      keyword: query.keyword || undefined,
      page: query.page,
      size: query.size,
    })
    items.value = res.items
    total.value = res.total
  } catch (e) {
    ElMessage.error(problemMessage(e))
  } finally {
    loading.value = false
  }
}

function search() {
  query.page = 1
  void load()
}

function openCreate() {
  createForm.productKey = query.productKey || products.value[0]?.productKey || ''
  createForm.name = ''
  createVisible.value = true
}

async function create() {
  if (!createForm.productKey || !createForm.name.trim()) {
    ElMessage.warning('请选择产品并填写设备名称')
    return
  }
  creating.value = true
  try {
    const res = await deviceApi.create(createForm.productKey, createForm.name.trim())
    createVisible.value = false
    showSecret(res.device.deviceKey, res.deviceSecret)
    await load()
  } catch (e) {
    ElMessage.error(problemMessage(e))
  } finally {
    creating.value = false
  }
}

function showSecret(deviceKey: string, value: string) {
  secret.deviceKey = deviceKey
  secret.value = value
  secretVisible.value = true
}

async function toggleStatus(row: DeviceView, enabled: boolean) {
  try {
    const updated = await deviceApi.update(row.deviceKey, row.name, enabled ? 'ENABLED' : 'DISABLED')
    row.status = updated.status
  } catch (e) {
    ElMessage.error(problemMessage(e))
  }
}

async function resetSecret(row: DeviceView) {
  try {
    await ElMessageBox.confirm(`重置后设备「${row.name}」需要使用新密钥重新连接，确定重置？`, '重置密钥', {
      type: 'warning',
    })
  } catch {
    return
  }
  try {
    const res = await deviceApi.resetSecret(row.deviceKey)
    showSecret(res.deviceKey, res.deviceSecret)
  } catch (e) {
    ElMessage.error(problemMessage(e))
  }
}

async function remove(row: DeviceView) {
  try {
    await ElMessageBox.confirm(`确定删除设备「${row.name}」？历史数据会随保留策略过期。`, '删除设备', {
      type: 'warning',
    })
  } catch {
    return
  }
  try {
    await deviceApi.remove(row.deviceKey)
    ElMessage.success('已删除')
    await load()
  } catch (e) {
    ElMessage.error(problemMessage(e))
  }
}

onMounted(async () => {
  await loadProducts()
  await load()
})
```

```java
package dev.conduit.e2e;

import java.io.IOException;
import java.io.UncheckedIOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.HashMap;
import java.util.Map;

/** 读取 deploy/.env，让测试与 Compose 使用同一套凭证。 */
final class DotEnv {

    private DotEnv() {
    }

    static Map<String, String> load(Path file) {
        try {
            Map<String, String> values = new HashMap<>();
            for (String line : Files.readAllLines(file)) {
                String trimmed = line.trim();
                if (trimmed.isEmpty() || trimmed.startsWith("#") || !trimmed.contains("=")) {
                    continue;
                }
                int eq = trimmed.indexOf('=');
                values.put(trimmed.substring(0, eq).trim(), trimmed.substring(eq + 1).trim());
            }
            return values;
        } catch (IOException e) {
            throw new UncheckedIOException("Cannot read " + file.toAbsolutePath(), e);
        }
    }
}
```

```java
package dev.conduit.e2e;

import com.influxdb.client.InfluxDBClient;
import com.influxdb.client.InfluxDBClientFactory;
import dev.conduit.simulator.ConduitApiClient;
import dev.conduit.simulator.DemoProvisioner;
import dev.conduit.simulator.DeviceCredentials;
import dev.conduit.simulator.DeviceSimulator;
import org.junit.jupiter.api.Test;

import java.net.URI;
import java.nio.file.Path;
import java.time.Duration;
import java.util.List;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.CompletableFuture;

import static org.assertj.core.api.Assertions.assertThat;
import static org.awaitility.Awaitility.await;

/**
 * 前置条件：已在仓库根目录执行 docker compose -f deploy/docker-compose.yml up -d。
 * 场景：10 台设备 × 50 条消息，发送过程中 kill 掉 conduit-server 再启动，最终 InfluxDB 恰好 500 条。
 */
class NoDataLossE2EIT {

    static final Path DEPLOY = Path.of("..", "..", "deploy").toAbsolutePath().normalize();
    static final int DEVICES = 10;
    static final int MESSAGES_PER_DEVICE = 50;

    final Map<String, String> env = DotEnv.load(DEPLOY.resolve(".env"));

    @Test
    void noDataLostWhenServerIsKilledMidStream() throws Exception {
        ConduitApiClient api = new ConduitApiClient(URI.create("http://localhost:8080"));
        api.login(env.get("CONDUIT_ADMIN_USERNAME"), env.get("CONDUIT_ADMIN_PASSWORD"));
        List<DeviceCredentials> devices = DemoProvisioner.provision(api, "e2e-" + UUID.randomUUID(), DEVICES);
        String productKey = devices.getFirst().productKey();

        DeviceSimulator simulator = new DeviceSimulator("localhost", 1883, Duration.ofMillis(100), MESSAGES_PER_DEVICE);
        CompletableFuture<Void> sending = CompletableFuture.runAsync(() -> {
            try {
                simulator.run(devices);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                throw new IllegalStateException(e);
            }
        });

        Thread.sleep(1_500);
        compose("kill", "conduit-server");
        Thread.sleep(3_000);
        compose("start", "conduit-server");

        sending.get();

        try (InfluxDBClient influx = InfluxDBClientFactory.create("http://localhost:8086",
                env.get("INFLUX_TOKEN").toCharArray(), "conduit", "telemetry")) {
            await().atMost(Duration.ofMinutes(4)).pollInterval(Duration.ofSeconds(5))
                    .untilAsserted(() -> assertThat(countUa(influx, productKey))
                            .isEqualTo((long) DEVICES * MESSAGES_PER_DEVICE));
        }
    }

    private static long countUa(InfluxDBClient influx, String productKey) {
        String flux = """
                from(bucket: "telemetry")
                  |> range(start: -1h)
                  |> filter(fn: (r) => r._measurement == "property" and r.product_key == "%s" and r._field == "Ua")
                  |> group()
                  |> count()
                """.formatted(productKey);
        return influx.getQueryApi().query(flux, "conduit").stream()
                .flatMap(t -> t.getRecords().stream())
                .mapToLong(r -> ((Number) r.getValue()).longValue())
                .sum();
    }

    private static void compose(String... args) throws Exception {
        String[] command = new String[4 + args.length];
        command[0] = "docker";
        command[1] = "compose";
        command[2] = "-f";
        command[3] = DEPLOY.resolve("docker-compose.yml").toString();
        System.arraycopy(args, 0, command, 4, args.length);
        Process process = new ProcessBuilder(command).inheritIO().start();
        assertThat(process.waitFor()).as("docker compose %s", String.join(" ", args)).isZero();
    }
}
```
