# Conduit V0.1 · P2 conduit-server Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现 conduit-server：多租户元数据、登录与 JWT、产品/设备管理接口、EMQX 鉴权回调，以及从 Redis Streams 消费、解码、写入 InfluxDB 与 Redis 最新值的数据流水线。

**Architecture:** 单个 Spring Boot 应用，按领域分包。PostgreSQL（MyBatis-Plus + 租户拦截器）存元数据；Redis 存最新值、在线状态、鉴权缓存（Lua 脚本保证按时间戳覆盖）；InfluxDB 存时序数据。流水线由 `PipelineWorker` 循环调用 `PipelineRunner`，`PipelineProcessor` 处理一批投递：可恢复错误不 ACK、不可恢复错误进死信。

**Tech Stack:** Java 21、Spring Boot 3.5.6（Web、Security、Validation、Data Redis、Actuator）、MyBatis-Plus 3.5.12、Flyway、PostgreSQL 16、jjwt 0.12.6、influxdb-client-java 7.2.0、Lombok、Testcontainers

前置条件：P1（`docs/superpowers/plans/2026-10-14-conduit-v0.1-p1-backend-core.md`）已完成。设计依据：`docs/superpowers/specs/2026-10-14-conduit-v0.1-design.md`。

## Global Constraints

- Java 21；Spring Boot 3.5.6；groupId `dev.conduit`；根包 `dev.conduit.server`。
- 单元测试 `*Test`，集成测试 `*IT`；`mvn verify` 两者都跑；集成测试需要 Docker Desktop。
- 镜像版本：`postgres:16.8`、`redis:7.4.2`、`influxdb:2.7.11`。
- 接口前缀 `/api/v1`；错误统一为 RFC 9457 `ProblemDetail`（400/401/403/404/409）。
- 所有业务表带 `tenant_id`；`tenant`、`app_user` 两表不走租户拦截器。
- `product_key`、`device_key` 16 位小写字母数字；`deviceSecret` 32 位；密钥存 `SHA-256(salt + secret)`（hex），salt 为 32 位 hex，常量时间比较；用户密码用 bcrypt。
- access token 15 分钟，refresh token 7 天，refresh token 放 HttpOnly Cookie `conduit_refresh`（`SameSite=Strict`，`Path=/api/v1/auth`）。
- Redis key：`conduit:latest:{deviceKey}`、`conduit:presence:{deviceKey}`、`conduit:auth:{deviceKey}`（TTL 5 分钟）；最新值 hash 中保留字段 `__last`。
- InfluxDB：bucket `telemetry`，measurement `property`，tag `tenant_id`/`product_key`/`device_key`，毫秒精度；数值测点统一写为 double。
- `ts` 有效范围 `[now − 30 天, now + 5 分钟]`。
- 死信原因码：`MALFORMED_ENVELOPE`、`MALFORMED_JSON`、`INVALID_PARAMS`、`UNKNOWN_DEVICE`、`DEVICE_DISABLED`、`PRODUCT_MISMATCH`、`TS_OUT_OF_RANGE`、`FIELD_TYPE_CONFLICT`、`UNSUPPORTED_TYPE`、`MAX_DELIVERY_EXCEEDED`。
- 流水线：消费组 `conduit-server`，每批 100 条，阻塞 1 秒；空闲 60 秒的待处理消息被认领，投递超过 5 次进死信；下游失败指数退避 1s→30s。
- 提交信息使用 Conventional Commits，不加任何 AI 署名。
- 命令在仓库根目录用 Git Bash 执行。

```sql
create table tenant (
    id         bigserial primary key,
    code       varchar(64)  not null unique,
    name       varchar(128) not null,
    status     varchar(16)  not null default 'ACTIVE',
    created_at timestamptz  not null default now()
);

create table app_user (
    id            bigserial primary key,
    tenant_id     bigint       not null references tenant (id),
    username      varchar(64)  not null unique,
    password_hash varchar(100) not null,
    role          varchar(32)  not null,
    status        varchar(16)  not null default 'ACTIVE',
    created_at    timestamptz  not null default now()
);

create table product (
    id          bigserial primary key,
    tenant_id   bigint       not null references tenant (id),
    product_key varchar(32)  not null unique,
    name        varchar(128) not null,
    description varchar(512),
    created_at  timestamptz  not null default now(),
    updated_at  timestamptz  not null default now()
);
create index idx_product_tenant on product (tenant_id);

create table device (
    id          bigserial primary key,
    tenant_id   bigint       not null references tenant (id),
    product_id  bigint       not null references product (id),
    device_key  varchar(32)  not null unique,
    name        varchar(128) not null,
    secret_hash varchar(64)  not null,
    secret_salt varchar(32)  not null,
    status      varchar(16)  not null default 'ENABLED',
    created_at  timestamptz  not null default now(),
    updated_at  timestamptz  not null default now()
);
create index idx_device_tenant_product on device (tenant_id, product_id);
```

```java
package dev.conduit.server.tenant;

import dev.conduit.server.device.DeviceIdentity;
import dev.conduit.server.device.DeviceMapper;
import dev.conduit.server.product.Product;
import dev.conduit.server.product.ProductMapper;
import dev.conduit.server.support.IntegrationTestBase;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;

import static org.assertj.core.api.Assertions.assertThat;
import static org.assertj.core.api.Assertions.assertThatThrownBy;

class TenantIsolationIT extends IntegrationTestBase {

    @Autowired
    ProductMapper products;

    @Autowired
    DeviceMapper devices;

    @Test
    void insertFillsTenantIdAndQueriesAreScopedToCurrentTenant() {
        long t1 = createTenant("t1-");
        long t2 = createTenant("t2-");
        String key = unique("pk");

        TenantContext.runAs(t1, () -> {
            Product p = new Product();
            p.setProductKey(key);
            p.setName("meter");
            products.insert(p);
        });

        assertThat(jdbc.queryForObject("select tenant_id from product where product_key = ?", Long.class, key))
                .isEqualTo(t1);
        assertThat(TenantContext.callAs(t1, () -> products.selectList(null)))
                .extracting(Product::getProductKey).containsExactly(key);
        assertThat(TenantContext.callAs(t2, () -> products.selectList(null))).isEmpty();
    }

    @Test
    void queryWithoutBoundTenantFails() {
        assertThatThrownBy(() -> products.selectList(null))
                .hasRootCauseInstanceOf(IllegalStateException.class);
    }
```

```java
package dev.conduit.server.tenant;

import java.util.function.Supplier;

/** 当前线程绑定的租户。HTTP 请求由 JwtAuthFilter 绑定，后台任务用 runAs/callAs 显式绑定。 */
public final class TenantContext {

    private static final ThreadLocal<Long> CURRENT = new ThreadLocal<>();

    private TenantContext() {
    }

    public static void set(Long tenantId) {
        CURRENT.set(tenantId);
    }

    public static Long get() {
        return CURRENT.get();
    }

    public static void clear() {
        CURRENT.remove();
    }

    public static long require() {
        Long id = CURRENT.get();
        if (id == null) {
            throw new IllegalStateException("No tenant bound to current thread");
        }
        return id;
    }

    public static <T> T callAs(long tenantId, Supplier<T> action) {
        Long previous = CURRENT.get();
        CURRENT.set(tenantId);
        try {
            return action.get();
        } finally {
            if (previous == null) {
                CURRENT.remove();
            } else {
                CURRENT.set(previous);
            }
        }
    }

    public static void runAs(long tenantId, Runnable action) {
        callAs(tenantId, () -> {
            action.run();
            return null;
        });
    }
}
```

```java
package dev.conduit.server.security;

import org.springframework.stereotype.Component;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.security.SecureRandom;
import java.util.HexFormat;

/**
 * 设备密钥哈希：SHA-256(salt + secret)。不用 bcrypt，避免重连风暴时鉴权 CPU 过载；
 * 密钥本身是 32 位随机串，熵足够，不需要慢哈希。
 */
@Component
public class SecretHasher {

    private final SecureRandom random = new SecureRandom();

    public String newSalt() {
        byte[] salt = new byte[16];
        random.nextBytes(salt);
        return HexFormat.of().formatHex(salt);
    }

    public String hash(String secret, String salt) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            return HexFormat.of().formatHex(digest.digest((salt + secret).getBytes(StandardCharsets.UTF_8)));
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException(e);
        }
    }

    public boolean matches(String secret, String salt, String expectedHash) {
        return MessageDigest.isEqual(
                hash(secret, salt).getBytes(StandardCharsets.UTF_8),
                expectedHash.getBytes(StandardCharsets.UTF_8));
    }
}
```

```java
@Component
public class AdminBootstrap implements ApplicationRunner {

    public static final String DEFAULT_TENANT_CODE = "default";
    public static final String ROLE_ADMIN = "ADMIN";

    private static final Logger log = LoggerFactory.getLogger(AdminBootstrap.class);

    private final BootstrapProperties props;
    private final TenantMapper tenants;
    private final AppUserMapper users;
    private final PasswordEncoder encoder;

    public AdminBootstrap(BootstrapProperties props, TenantMapper tenants, AppUserMapper users,

        this.tenants = tenants;
        this.users = users;
        this.encoder = encoder;
    }

    @Override
    public void run(ApplicationArguments args) {
        if (props.adminUsername() == null || props.adminUsername().isBlank()
                || props.adminPassword() == null || props.adminPassword().isBlank()) {
            throw new IllegalStateException(
                    "conduit.bootstrap.admin-username and conduit.bootstrap.admin-password must be set");
        }

        Tenant tenant = tenants.selectOne(new LambdaQueryWrapper<Tenant>().eq(Tenant::getCode, DEFAULT_TENANT_CODE));

        boolean adminExists = users.exists(
                new LambdaQueryWrapper<AppUser>().eq(AppUser::getUsername, props.adminUsername()));

            AppUser admin = new AppUser();
            admin.setTenantId(tenant.getId());
            admin.setUsername(props.adminUsername());
            admin.setPasswordHash(encoder.encode(props.adminPassword()));
            admin.setRole(ROLE_ADMIN);
```

```java
package dev.conduit.server.auth;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.JwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.stereotype.Component;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.time.Clock;
import java.time.Duration;
import java.time.Instant;
import java.util.Date;

@Component
public class JwtService {

    public enum TokenType { ACCESS, REFRESH }

    private final JwtProperties props;
    private final Clock clock;
    private final SecretKey key;

    public JwtService(JwtProperties props, Clock clock) {
        if (props.jwtSecret() == null || props.jwtSecret().getBytes(StandardCharsets.UTF_8).length < 32) {
            throw new IllegalStateException("conduit.auth.jwt-secret must be at least 32 bytes");
        }
        this.props = props;
        this.clock = clock;
        this.key = Keys.hmacShaKeyFor(props.jwtSecret().getBytes(StandardCharsets.UTF_8));
    }

    public String issue(AuthPrincipal principal, TokenType type) {
        Instant now = clock.instant();
        Duration ttl = type == TokenType.ACCESS ? props.accessTtl() : props.refreshTtl();
        return Jwts.builder()
                .subject(Long.toString(principal.userId()))
                .claim("tid", principal.tenantId())
                .claim("usr", principal.username())
                .claim("role", principal.role())
                .claim("typ", type.name())
                .issuedAt(Date.from(now))
                .expiration(Date.from(now.plus(ttl)))
                .signWith(key)
                .compact();
    }

    public AuthPrincipal parse(String token, TokenType expected) {
        Claims claims = Jwts.parser()
                .verifyWith(key)
                .clock(() -> Date.from(clock.instant()))
                .build()
                .parseSignedClaims(token)
                .getPayload();
        if (!expected.name().equals(claims.get("typ", String.class))) {
            throw new JwtException("Unexpected token type");
        }
        return new AuthPrincipal(
                Long.parseLong(claims.getSubject()),
                ((Number) claims.get("tid")).longValue(),
                claims.get("usr", String.class),
                claims.get("role", String.class));
    }
}
```

```java
package dev.conduit.server.auth;

import dev.conduit.server.tenant.TenantContext;
import io.jsonwebtoken.JwtException;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.http.HttpHeaders;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.util.List;

/** 解析 Bearer access token，设置认证信息并绑定租户；token 无效时保持未认证，由入口点返回 401。 */
public class JwtAuthFilter extends OncePerRequestFilter {

    private final JwtService jwt;

    public JwtAuthFilter(JwtService jwt) {
        this.jwt = jwt;
    }

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain chain)
            throws ServletException, IOException {
        String header = request.getHeader(HttpHeaders.AUTHORIZATION);
        if (header != null && header.startsWith("Bearer ")) {
            try {
                AuthPrincipal principal = jwt.parse(header.substring(7), JwtService.TokenType.ACCESS);
                var authentication = new UsernamePasswordAuthenticationToken(principal, null,
                        List.of(new SimpleGrantedAuthority("ROLE_" + principal.role())));
                SecurityContextHolder.getContext().setAuthentication(authentication);
                TenantContext.set(principal.tenantId());
            } catch (JwtException | IllegalArgumentException ignored) {
                // 保持未认证
            }
        }
        try {
            chain.doFilter(request, response);
        } finally {
            TenantContext.clear();
        }
    }
}
```

```java
package dev.conduit.server.auth;

import com.fasterxml.jackson.databind.ObjectMapper;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;

import java.io.IOException;

@Configuration(proxyBeanMethods = false)
public class SecurityConfig {

    @Bean
    SecurityFilterChain securityFilterChain(HttpSecurity http, JwtService jwt, ObjectMapper objectMapper)
            throws Exception {
        http
                // 无会话 + refresh Cookie 为 SameSite=Strict 且只发往 /api/v1/auth，关闭 CSRF
                .csrf(c -> c.disable())
                .httpBasic(b -> b.disable())
                .formLogin(f -> f.disable())
                .sessionManagement(s -> s.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .authorizeHttpRequests(a -> a
                        .requestMatchers("/api/v1/auth/login", "/api/v1/auth/refresh", "/api/v1/auth/logout").permitAll()
                        .requestMatchers("/internal/**").permitAll()
                        .requestMatchers("/actuator/health", "/actuator/health/**", "/error").permitAll()
                        .anyRequest().authenticated())
                .exceptionHandling(e -> e
                        .authenticationEntryPoint((req, res, ex) ->
                                writeProblem(res, objectMapper, HttpStatus.UNAUTHORIZED, "未认证或登录已过期"))
                        .accessDeniedHandler((req, res, ex) ->
                                writeProblem(res, objectMapper, HttpStatus.FORBIDDEN, "无权限")))
                .addFilterBefore(new JwtAuthFilter(jwt), UsernamePasswordAuthenticationFilter.class);
        return http.build();
    }

    private static void writeProblem(HttpServletResponse response, ObjectMapper objectMapper,
                                     HttpStatus status, String detail) throws IOException {
        response.setStatus(status.value());
        response.setContentType("application/problem+json");
        objectMapper.writeValue(response.getOutputStream(), ProblemDetail.forStatusAndDetail(status, detail));
    }
}
```

```java
package dev.conduit.server.auth;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import dev.conduit.server.auth.AuthDtos.LoginRequest;
import dev.conduit.server.auth.AuthDtos.TokenResponse;
import dev.conduit.server.auth.AuthDtos.UserView;
import dev.conduit.server.web.ApiException;
import io.jsonwebtoken.JwtException;
import jakarta.validation.Valid;
import org.springframework.http.HttpHeaders;
import org.springframework.http.ResponseCookie;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.web.bind.annotation.CookieValue;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import java.time.Duration;

@RestController
@RequestMapping("/api/v1/auth")
public class AuthController {

    static final String REFRESH_COOKIE = "conduit_refresh";
    private static final String ACTIVE = "ACTIVE";

    private final AppUserMapper users;
    private final PasswordEncoder encoder;
    private final JwtService jwt;
    private final JwtProperties props;

    public AuthController(AppUserMapper users, PasswordEncoder encoder, JwtService jwt, JwtProperties props) {
        this.users = users;
        this.encoder = encoder;
        this.jwt = jwt;
        this.props = props;
    }

    @PostMapping("/login")
    public ResponseEntity<TokenResponse> login(@Valid @RequestBody LoginRequest request) {
        AppUser user = users.selectOne(new LambdaQueryWrapper<AppUser>().eq(AppUser::getUsername, request.username()));
        if (user == null || !ACTIVE.equals(user.getStatus())
                || !encoder.matches(request.password(), user.getPasswordHash())) {
            throw ApiException.unauthorized("用户名或密码错误");
        }
        return tokens(principalOf(user));
    }

    @PostMapping("/refresh")
    public ResponseEntity<TokenResponse> refresh(
            @CookieValue(name = REFRESH_COOKIE, required = false) String refreshToken) {
        if (refreshToken == null || refreshToken.isBlank()) {
            throw ApiException.unauthorized("缺少刷新凭证");
        }
        AuthPrincipal principal;
        try {
            principal = jwt.parse(refreshToken, JwtService.TokenType.REFRESH);
        } catch (JwtException | IllegalArgumentException e) {
            throw ApiException.unauthorized("刷新凭证无效");
        }
        AppUser user = users.selectById(principal.userId());
        if (user == null || !ACTIVE.equals(user.getStatus())) {
            throw ApiException.unauthorized("用户不可用");
        }
        return tokens(principalOf(user));
    }

    @PostMapping("/logout")
    public ResponseEntity<Void> logout() {
        return ResponseEntity.noContent()
                .header(HttpHeaders.SET_COOKIE, refreshCookie("", Duration.ZERO).toString())
                .build();
    }

    @GetMapping("/me")
    public UserView me(@AuthenticationPrincipal AuthPrincipal principal) {
        return new UserView(principal.username(), principal.role());
    }

    private ResponseEntity<TokenResponse> tokens(AuthPrincipal principal) {
        String access = jwt.issue(principal, JwtService.TokenType.ACCESS);
        String refresh = jwt.issue(principal, JwtService.TokenType.REFRESH);
        return ResponseEntity.ok()
                .header(HttpHeaders.SET_COOKIE, refreshCookie(refresh, props.refreshTtl()).toString())
                .body(new TokenResponse(access, props.accessTtl().toSeconds(),
                        new UserView(principal.username(), principal.role())));
    }

    private ResponseCookie refreshCookie(String value, Duration maxAge) {
        return ResponseCookie.from(REFRESH_COOKIE, value)
                .httpOnly(true)
                .secure(props.cookieSecure())
                .sameSite("Strict")
                .path("/api/v1/auth")
                .maxAge(maxAge)
                .build();
    }

    private static AuthPrincipal principalOf(AppUser user) {
        return new AuthPrincipal(user.getId(), user.getTenantId(), user.getUsername(), user.getRole());
    }
}
```

`backend/conduit-server/src/main/resources/redis/latest-update.lua`：

```lua
-- KEYS[1] = conduit:latest:{deviceKey}
-- ARGV[1] = 本次数据的 ts（毫秒）
-- ARGV[2..] = 测点名, {"v":...,"ts":...} 交替
-- 逐测点比较 ts：只有不早于当前值的数据才覆盖，补传的旧数据不会盖掉新值。
local ts = tonumber(ARGV[1])
local updated = 0
for i = 2, #ARGV, 2 do
  local current = redis.call('HGET', KEYS[1], ARGV[i])
  if (not current) or ts >= tonumber(cjson.decode(current)['ts']) then
    redis.call('HSET', KEYS[1], ARGV[i], ARGV[i + 1])
    updated = updated + 1
  end
end
local last = redis.call('HGET', KEYS[1], '__last')
if (not last) or ts > tonumber(last) then
  redis.call('HSET', KEYS[1], '__last', ARGV[1])
end
return updated
```

```java
package dev.conduit.server.telemetry;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.core.io.ClassPathResource;
import org.springframework.data.redis.core.RedisCallback;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.data.redis.core.script.RedisScript;
import org.springframework.stereotype.Component;

import java.io.UncheckedIOException;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;

@Component
public class LatestValueStore {

    static final String LAST_FIELD = "__last";
    private static final RedisScript<Long> UPDATE =
            RedisScript.of(new ClassPathResource("redis/latest-update.lua"), Long.class);

    private final StringRedisTemplate redis;
    private final ObjectMapper objectMapper;

    public LatestValueStore(StringRedisTemplate redis, ObjectMapper objectMapper) {
        this.redis = redis;
        this.objectMapper = objectMapper;
    }

    public static String key(String deviceKey) {
        return "conduit:latest:" + deviceKey;
    }

    public void update(String deviceKey, long ts, Map<String, Object> params) {
        List<String> args = new ArrayList<>(1 + params.size() * 2);
        args.add(Long.toString(ts));
        params.forEach((name, value) -> {
            args.add(name);
            args.add(toJson(Map.of("v", value, "ts", ts)));
        });
        redis.execute(UPDATE, List.of(key(deviceKey)), args.toArray());
    }

    public Map<String, LatestValue> get(String deviceKey) {
        Map<Object, Object> entries = redis.opsForHash().entries(key(deviceKey));
        Map<String, LatestValue> result = new TreeMap<>();
        entries.forEach((field, json) -> {
            String name = (String) field;
            if (!LAST_FIELD.equals(name)) {
                result.put(name, parse((String) json));
            }
        });
        return result;
    }

    public Map<String, Long> lastReportTimes(List<String> deviceKeys) {
        if (deviceKeys.isEmpty()) {
            return Map.of();
        }
        byte[] lastField = LAST_FIELD.getBytes(StandardCharsets.UTF_8);
        List<Object> values = redis.executePipelined((RedisCallback<Object>) connection -> {
            for (String dk : deviceKeys) {
                connection.hashCommands().hGet(key(dk).getBytes(StandardCharsets.UTF_8), lastField);
            }
            return null;
        });
        Map<String, Long> result = new LinkedHashMap<>();
        for (int i = 0; i < deviceKeys.size(); i++) {
            Object value = values.get(i);
            if (value != null) {
                result.put(deviceKeys.get(i), Long.parseLong((String) value));
            }
        }
        return result;
    }

    public void delete(String deviceKey) {
        redis.delete(key(deviceKey));
    }

    private LatestValue parse(String json) {
        try {
            JsonNode node = objectMapper.readTree(json);
            return new LatestValue(objectMapper.treeToValue(node.get("v"), Object.class), node.get("ts").asLong());
        } catch (JsonProcessingException e) {
            throw new UncheckedIOException(e);
        }
    }

    private String toJson(Object value) {
        try {
            return objectMapper.writeValueAsString(value);
        } catch (JsonProcessingException e) {
            throw new UncheckedIOException(e);
        }
    }
}
```

```java
package dev.conduit.server.presence;

import org.springframework.core.io.ClassPathResource;
import org.springframework.data.redis.core.RedisCallback;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.data.redis.core.script.RedisScript;
import org.springframework.stereotype.Component;

import java.nio.charset.StandardCharsets;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Optional;

@Component
public class PresenceStore {

    private static final RedisScript<Long> APPLY =
            RedisScript.of(new ClassPathResource("redis/presence-apply.lua"), Long.class);

    private final StringRedisTemplate redis;

    public PresenceStore(StringRedisTemplate redis) {
        this.redis = redis;
    }

    public static String key(String deviceKey) {
        return "conduit:presence:" + deviceKey;
    }

    public boolean apply(String deviceKey, boolean online, long eventTime, String clientIp) {
        Long applied = redis.execute(APPLY, List.of(key(deviceKey)),
                Boolean.toString(online), Long.toString(eventTime), clientIp == null ? "" : clientIp);
        return applied != null && applied == 1L;
    }

    public Optional<Presence> get(String deviceKey) {
        return Optional.ofNullable(toPresence(redis.opsForHash().entries(key(deviceKey))));
    }

    public Map<String, Presence> getAll(List<String> deviceKeys) {
        if (deviceKeys.isEmpty()) {
            return Map.of();
        }
        List<Object> results = redis.executePipelined((RedisCallback<Object>) connection -> {
            for (String dk : deviceKeys) {
                connection.hashCommands().hGetAll(key(dk).getBytes(StandardCharsets.UTF_8));
            }
            return null;
        });
        Map<String, Presence> presence = new LinkedHashMap<>();
        for (int i = 0; i < deviceKeys.size(); i++) {
            @SuppressWarnings("unchecked")
            Presence p = toPresence((Map<Object, Object>) results.get(i));
            if (p != null) {
                presence.put(deviceKeys.get(i), p);
            }
        }
        return presence;
    }

    public void delete(String deviceKey) {
        redis.delete(key(deviceKey));
    }

    private static Presence toPresence(Map<Object, Object> hash) {
        if (hash == null || hash.isEmpty()) {
            return null;
        }
        return new Presence(
                Boolean.parseBoolean((String) hash.get("online")),
                Long.parseLong((String) hash.get("since")),
                (String) hash.getOrDefault("clientIp", ""));
    }
}
```

```java
package dev.conduit.server.device;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Component;

import java.io.UncheckedIOException;
import java.time.Duration;
import java.util.Optional;

@Component
public class DeviceIdentityCache {

    static final Duration TTL = Duration.ofMinutes(5);

    private final StringRedisTemplate redis;
    private final ObjectMapper objectMapper;

    public DeviceIdentityCache(StringRedisTemplate redis, ObjectMapper objectMapper) {
        this.redis = redis;
        this.objectMapper = objectMapper;
    }

    public static String key(String deviceKey) {
        return "conduit:auth:" + deviceKey;
    }

    public Optional<DeviceIdentity> get(String deviceKey) {
        String json = redis.opsForValue().get(key(deviceKey));
        if (json == null) {
            return Optional.empty();
        }
        try {
            return Optional.of(objectMapper.readValue(json, DeviceIdentity.class));
        } catch (JsonProcessingException e) {
            throw new UncheckedIOException(e);
        }
    }

    public void put(DeviceIdentity identity) {
        try {
            redis.opsForValue().set(key(identity.deviceKey()), objectMapper.writeValueAsString(identity), TTL);
        } catch (JsonProcessingException e) {
            throw new UncheckedIOException(e);
        }
    }

    public void evict(String deviceKey) {
        redis.delete(key(deviceKey));
    }
}
```

```java
package dev.conduit.server.device;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.dao.DataAccessException;
import org.springframework.stereotype.Component;

import java.util.Optional;

/** 设备身份查询：缓存优先，Redis 不可用时降级直查 PostgreSQL。 */
@Component
public class DeviceIdentityLookup {

    private static final Logger log = LoggerFactory.getLogger(DeviceIdentityLookup.class);

    private final DeviceIdentityCache cache;
    private final DeviceMapper devices;

    public DeviceIdentityLookup(DeviceIdentityCache cache, DeviceMapper devices) {
        this.cache = cache;
        this.devices = devices;
    }

    public Optional<DeviceIdentity> find(String deviceKey) {
        try {
            Optional<DeviceIdentity> cached = cache.get(deviceKey);
            if (cached.isPresent()) {
                return cached;
            }
        } catch (DataAccessException e) {
            log.warn("Identity cache unavailable, falling back to database: {}", e.getMessage());
        }

        DeviceIdentity identity = devices.findIdentity(deviceKey);
        if (identity == null) {
            return Optional.empty();
        }
        try {
            cache.put(identity);
        } catch (DataAccessException e) {
            log.warn("Failed to fill identity cache: {}", e.getMessage());
        }
        return Optional.of(identity);
    }
}
```

```java
package dev.conduit.server.device;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import dev.conduit.server.device.DeviceDtos.CreateDeviceRequest;
import dev.conduit.server.device.DeviceDtos.DeviceCreated;
import dev.conduit.server.device.DeviceDtos.DeviceView;
import dev.conduit.server.device.DeviceDtos.SecretView;
import dev.conduit.server.device.DeviceDtos.UpdateDeviceRequest;
import dev.conduit.server.presence.Presence;
import dev.conduit.server.presence.PresenceStore;
import dev.conduit.server.product.Product;
import dev.conduit.server.product.ProductMapper;
import dev.conduit.server.product.ProductService;
import dev.conduit.server.security.KeyGenerator;
import dev.conduit.server.security.SecretHasher;
import dev.conduit.server.telemetry.LatestValueStore;
import dev.conduit.server.web.ApiException;
import dev.conduit.server.web.PageView;
import org.springframework.stereotype.Service;

import java.time.Clock;
import java.time.OffsetDateTime;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

@Service
public class DeviceService {

    private final DeviceMapper devices;
    private final ProductMapper products;
    private final ProductService productService;
    private final KeyGenerator keys;
    private final SecretHasher hasher;
    private final DeviceIdentityCache identityCache;
    private final LatestValueStore latest;
    private final PresenceStore presence;
    private final Clock clock;

    public DeviceService(DeviceMapper devices, ProductMapper products, ProductService productService,
                         KeyGenerator keys, SecretHasher hasher, DeviceIdentityCache identityCache,
                         LatestValueStore latest, PresenceStore presence, Clock clock) {
        this.devices = devices;
        this.products = products;
        this.productService = productService;
        this.keys = keys;
        this.hasher = hasher;
        this.identityCache = identityCache;
        this.latest = latest;
        this.presence = presence;
        this.clock = clock;
    }

    public DeviceCreated create(CreateDeviceRequest request) {
        Product product = productService.require(request.productKey());
        String secret = keys.deviceSecret();
        String salt = hasher.newSalt();

        Device d = new Device();
        d.setProductId(product.getId());
        d.setDeviceKey(keys.deviceKey());
        d.setName(request.name());
        d.setSecretSalt(salt);
        d.setSecretHash(hasher.hash(secret, salt));
        d.setStatus(Device.ENABLED);
        devices.insert(d);

        Device saved = devices.selectById(d.getId());
        return new DeviceCreated(toViews(List.of(saved)).getFirst(), secret);
    }

    public PageView<DeviceView> list(String productKey, String keyword, int page, int size) {
        LambdaQueryWrapper<Device> query = new LambdaQueryWrapper<Device>().orderByDesc(Device::getId);
        if (productKey != null && !productKey.isBlank()) {
            query.eq(Device::getProductId, productService.require(productKey).getId());
        }
        if (keyword != null && !keyword.isBlank()) {
            query.and(q -> q.like(Device::getName, keyword).or().like(Device::getDeviceKey, keyword));
        }
        Page<Device> result = devices.selectPage(new Page<>(page, size), query);
        return new PageView<>(toViews(result.getRecords()), result.getTotal(), page, size);
    }

    public DeviceView get(String deviceKey) {
        return toViews(List.of(require(deviceKey))).getFirst();
    }

    public DeviceView update(String deviceKey, UpdateDeviceRequest request) {
        Device d = require(deviceKey);
        boolean statusChanged = !d.getStatus().equals(request.status());
        d.setName(request.name());
        d.setStatus(request.status());
        d.setUpdatedAt(OffsetDateTime.now(clock));
        devices.updateById(d);
        if (statusChanged) {
            identityCache.evict(deviceKey);
        }
        return toViews(List.of(d)).getFirst();
    }

    public SecretView resetSecret(String deviceKey) {
        Device d = require(deviceKey);
        String secret = keys.deviceSecret();
        String salt = hasher.newSalt();
        d.setSecretSalt(salt);
        d.setSecretHash(hasher.hash(secret, salt));
        d.setUpdatedAt(OffsetDateTime.now(clock));
        devices.updateById(d);
        identityCache.evict(deviceKey);
        return new SecretView(deviceKey, secret);
    }

    public void delete(String deviceKey) {
        Device d = require(deviceKey);
        devices.deleteById(d.getId());
        identityCache.evict(deviceKey);
        latest.delete(deviceKey);
        presence.delete(deviceKey);
    }

    public Device require(String deviceKey) {
        Device d = devices.selectOne(new LambdaQueryWrapper<Device>().eq(Device::getDeviceKey, deviceKey));
        if (d == null) {
            throw ApiException.notFound("设备不存在: " + deviceKey);
        }
        return d;
    }

    private List<DeviceView> toViews(List<Device> list) {
        if (list.isEmpty()) {
            return List.of();
        }
        List<String> deviceKeys = list.stream().map(Device::getDeviceKey).toList();
        Map<Long, String> productKeys = products.selectBatchIds(list.stream().map(Device::getProductId).distinct().toList())
                .stream().collect(Collectors.toMap(Product::getId, Product::getProductKey));
        Map<String, Presence> presenceByDevice = presence.getAll(deviceKeys);
        Map<String, Long> lastReports = latest.lastReportTimes(deviceKeys);

        return list.stream().map(d -> {
            Presence p = presenceByDevice.get(d.getDeviceKey());
            return new DeviceView(d.getDeviceKey(), d.getName(), productKeys.get(d.getProductId()), d.getStatus(),
                    p != null && p.online(), lastReports.get(d.getDeviceKey()), d.getCreatedAt());
        }).toList();
    }
}
```

```java
package dev.conduit.server.mqttauth;

import dev.conduit.server.device.DeviceIdentity;
import dev.conduit.server.device.DeviceIdentityLookup;
import dev.conduit.server.security.SecretHasher;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.util.Optional;

@Service
public class MqttAuthService {

    public record Decision(boolean allow, boolean superuser) {
        public static final Decision DENY = new Decision(false, false);
    }

    static final String INTERNAL_CLIENT_PREFIX = "conduit-";
    private static final Logger log = LoggerFactory.getLogger(MqttAuthService.class);

    private final MqttAuthProperties props;
    private final DeviceIdentityLookup lookup;
    private final SecretHasher hasher;

    public MqttAuthService(MqttAuthProperties props, DeviceIdentityLookup lookup, SecretHasher hasher) {
        this.props = props;
        this.lookup = lookup;
        this.hasher = hasher;
    }

    public Decision authenticate(String clientId, String username, String password) {
        if (clientId == null || username == null || password == null) {
            return Decision.DENY;
        }
        if (username.equals(props.connectorUsername())) {
            boolean ok = props.connectorPassword() != null && !props.connectorPassword().isBlank()
                    && constantEquals(password, props.connectorPassword())
                    && clientId.startsWith(INTERNAL_CLIENT_PREFIX);
            return ok ? new Decision(true, true) : deny("connector credentials rejected", clientId);
        }
        if (!clientId.equals(username)) {
            return deny("clientId does not match username", clientId);
        }
        Optional<DeviceIdentity> identity = lookup.find(username);
        if (identity.isEmpty()) {
            return deny("unknown device", clientId);
        }
        DeviceIdentity d = identity.get();
        if (!d.enabled()) {
            return deny("device disabled", clientId);
        }
        if (!hasher.matches(password, d.secretSalt(), d.secretHash())) {
            return deny("wrong secret", clientId);
        }
        return new Decision(true, false);
    }

    static boolean constantEquals(String a, String b) {
        return MessageDigest.isEqual(a.getBytes(StandardCharsets.UTF_8), b.getBytes(StandardCharsets.UTF_8));
    }

    private static Decision deny(String reason, String clientId) {
        log.debug("MQTT auth denied for {}: {}", clientId, reason);
        return Decision.DENY;
    }
}
```

```java
package dev.conduit.server.mqttauth;

import dev.conduit.server.mqttauth.MqttAuthService.Decision;
import dev.conduit.server.web.ApiException;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;
import org.springframework.web.bind.annotation.RestController;

import java.util.Map;

/** EMQX HTTP 鉴权回调。只返回 200 + allow/deny；内部 token 不对返回 401（EMQX 视为失败并拒绝连接）。 */
@RestController
public class MqttAuthController {

    public record MqttAuthRequest(String clientid, String username, String password) {
    }

    private final MqttAuthService service;
    private final MqttAuthProperties props;

    public MqttAuthController(MqttAuthService service, MqttAuthProperties props) {
        this.service = service;
        this.props = props;
    }

    @PostMapping("/internal/mqtt/auth")
    public Map<String, Object> authenticate(
            @RequestHeader(value = "X-Conduit-Internal-Token", required = false) String token,
            @RequestBody MqttAuthRequest request) {
        if (token == null || props.token() == null || props.token().isBlank()
                || !MqttAuthService.constantEquals(token, props.token())) {
            throw ApiException.unauthorized("invalid internal token");
        }
        Decision decision = service.authenticate(request.clientid(), request.username(), request.password());
        if (!decision.allow()) {
            return Map.of("result", "deny");
        }
        return Map.of("result", "allow", "is_superuser", decision.superuser());
    }
}
```

```java
package dev.conduit.server.pipeline;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.stereotype.Component;

import java.io.IOException;
import java.time.Clock;
import java.time.Duration;
import java.util.Iterator;
import java.util.LinkedHashMap;
import java.util.Map;

/**
 * 解码 Conduit 标准属性格式：{"id":..., "ts":毫秒, "params":{测点: 数值|字符串|布尔}}。
 * 数值统一转为 double，避免同一测点先报整数后报小数在 InfluxDB 中产生类型冲突。
 */
@Component
public class PropertyDecoder {

    static final Duration MAX_PAST = Duration.ofDays(30);
    static final Duration MAX_FUTURE = Duration.ofMinutes(5);

    private final ObjectMapper objectMapper;
    private final Clock clock;

    public PropertyDecoder(ObjectMapper objectMapper, Clock clock) {
        this.objectMapper = objectMapper;
        this.clock = clock;
    }

    public DecodedProperty decode(byte[] payload, long receivedAt) {
        JsonNode root;
        try {
            root = objectMapper.readTree(payload);
        } catch (IOException e) {
            throw new DecodeException(DeadReason.MALFORMED_JSON, e.getOriginalMessage());
        }
        if (root == null || !root.isObject()) {
            throw new DecodeException(DeadReason.MALFORMED_JSON, "payload is not a JSON object");
        }
        return new DecodedProperty(timestamp(root.get("ts"), receivedAt), params(root.get("params")));
    }

    private long timestamp(JsonNode node, long receivedAt) {
        if (node == null || node.isNull()) {
            return receivedAt;
        }
        if (!node.isIntegralNumber() || !node.canConvertToLong()) {
            throw new DecodeException(DeadReason.TS_OUT_OF_RANGE, "ts must be integer milliseconds");
        }
        long ts = node.asLong();
        long now = clock.millis();
        if (ts < now - MAX_PAST.toMillis() || ts > now + MAX_FUTURE.toMillis()) {
            throw new DecodeException(DeadReason.TS_OUT_OF_RANGE, "ts " + ts + " outside allowed window");
        }
        return ts;
    }

    private static Map<String, Object> params(JsonNode node) {
        if (node == null || !node.isObject() || node.isEmpty()) {
            throw new DecodeException(DeadReason.INVALID_PARAMS, "params must be a non-empty object");
        }
        Map<String, Object> values = new LinkedHashMap<>();
        Iterator<Map.Entry<String, JsonNode>> fields = node.fields();
        while (fields.hasNext()) {
            Map.Entry<String, JsonNode> field = fields.next();
            String name = field.getKey();
            if (name.isBlank() || name.startsWith("__")) {
                throw new DecodeException(DeadReason.INVALID_PARAMS, "invalid field name '" + name + "'");
            }
            JsonNode v = field.getValue();
            Object value;
            if (v.isNumber()) {
                value = v.doubleValue();
            } else if (v.isTextual()) {
                value = v.textValue();
            } else if (v.isBoolean()) {
                value = v.booleanValue();
            } else {
                throw new DecodeException(DeadReason.INVALID_PARAMS, "unsupported value type for '" + name + "'");
            }
            values.put(name, value);
        }
        return values;
    }
}
```

```java
package dev.conduit.server.telemetry;

import com.influxdb.client.InfluxDBClient;
import com.influxdb.client.WriteApiBlocking;
import com.influxdb.client.domain.WritePrecision;
import com.influxdb.client.write.Point;
import com.influxdb.exceptions.InfluxException;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;

@Component
public class TelemetryWriter {

    static final String MEASUREMENT = "property";

    private final InfluxDBClient client;
    private final InfluxProperties props;

    public TelemetryWriter(InfluxDBClient client, InfluxProperties props) {
        this.client = client;
        this.props = props;
    }

    public WriteResult write(List<TelemetryPoint> points) {
        if (points.isEmpty()) {
            return WriteResult.ok();
        }
        WriteApiBlocking api = client.getWriteApiBlocking();
        try {
            api.writePoints(props.bucket(), props.org(), points.stream().map(TelemetryWriter::toPoint).toList());
            return WriteResult.ok();
        } catch (InfluxException e) {
            if (!isDataRejection(e)) {
                throw new TelemetryUnavailableException(e);
            }
        } catch (RuntimeException e) {
            throw new TelemetryUnavailableException(e);
        }

        // 整批被拒（通常是字段类型冲突）：逐条重写，找出真正有问题的点。重写成功的点是幂等的。
        List<WriteResult.Rejected> rejected = new ArrayList<>();
        for (int i = 0; i < points.size(); i++) {
            try {
                api.writePoint(props.bucket(), props.org(), toPoint(points.get(i)));
            } catch (InfluxException e) {
                if (!isDataRejection(e)) {
                    throw new TelemetryUnavailableException(e);
                }
                rejected.add(new WriteResult.Rejected(i, e.getMessage()));
            } catch (RuntimeException e) {
                throw new TelemetryUnavailableException(e);
            }
        }
        return new WriteResult(rejected);
    }

    /** 只有 400/422 视为数据问题；401/403/404 是配置问题，按不可用处理以免整批进死信。 */
    private static boolean isDataRejection(InfluxException e) {
        return e.status() == 400 || e.status() == 422;
    }

    static Point toPoint(TelemetryPoint p) {
        Point point = Point.measurement(MEASUREMENT)
                .addTag("tenant_id", Long.toString(p.tenantId()))
                .addTag("product_key", p.productKey())
                .addTag("device_key", p.deviceKey())
                .time(p.ts(), WritePrecision.MS);
        p.fields().forEach((name, value) -> {
            switch (value) {
                case Number n -> point.addField(name, n.doubleValue());
                case Boolean b -> point.addField(name, b);
                case String s -> point.addField(name, s);
                default -> throw new IllegalArgumentException("Unsupported field type for " + name);
            }
        });
        return point;
    }
}
```

```java
package dev.conduit.server.presence;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import dev.conduit.common.message.MessageType;
import dev.conduit.common.message.RawMessage;
import dev.conduit.server.pipeline.DeadReason;
import dev.conduit.server.pipeline.DecodeException;
import org.springframework.stereotype.Component;

import java.io.IOException;

/** 处理 EMQX $SYS 上下线事件：以事件自带时间为准更新在线状态。 */
@Component
public class PresenceHandler {

    private final ObjectMapper objectMapper;
    private final PresenceStore store;

    public PresenceHandler(ObjectMapper objectMapper, PresenceStore store) {
        this.objectMapper = objectMapper;
        this.store = store;
    }

    public void handle(RawMessage message) {
        JsonNode root;
        try {
            root = objectMapper.readTree(message.payload());
        } catch (IOException e) {
            throw new DecodeException(DeadReason.MALFORMED_JSON, e.getMessage());
        }
        if (root == null || !root.isObject()) {
            throw new DecodeException(DeadReason.MALFORMED_JSON, "presence payload is not a JSON object");
        }
        boolean online = message.type() == MessageType.PRESENCE_CONNECTED;
        JsonNode time = root.get(online ? "connected_at" : "disconnected_at");
        long eventTime = time != null && time.canConvertToLong() ? time.asLong() : message.receivedAt();
        String clientIp = root.path("ipaddress").asText("");
        store.apply(message.deviceKey(), online, eventTime, clientIp);
    }
}
```

```java
package dev.conduit.server.pipeline;

import dev.conduit.common.bus.Delivery;
import dev.conduit.common.bus.MessageBus;
import dev.conduit.common.message.RawMessage;
import dev.conduit.server.device.DeviceIdentity;
import dev.conduit.server.device.DeviceIdentityLookup;
import dev.conduit.server.presence.PresenceHandler;
import dev.conduit.server.telemetry.LatestValueStore;
import dev.conduit.server.telemetry.TelemetryPoint;
import dev.conduit.server.telemetry.TelemetryWriter;
import dev.conduit.server.telemetry.WriteResult;
import io.micrometer.core.instrument.MeterRegistry;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.Set;
import java.util.stream.Collectors;

/**
 * 处理一批投递。不可恢复的错误进死信（并 ACK）；下游不可用时直接抛出，未处理的投递保持未 ACK，
 * 之后由认领机制重新投递。重复投递由 InfluxDB 同 ts 覆盖与最新值按 ts 比较吸收。
 */
@Component
public class PipelineProcessor {

    private static final Logger log = LoggerFactory.getLogger(PipelineProcessor.class);

    private record PendingProperty(Delivery delivery, TelemetryPoint point) {
    }

    private final MessageBus bus;
    private final PipelineProperties props;
    private final DeviceIdentityLookup identities;
    private final PropertyDecoder decoder;
    private final PresenceHandler presenceHandler;
    private final TelemetryWriter writer;
    private final LatestValueStore latest;
    private final MeterRegistry meters;

    public PipelineProcessor(MessageBus bus, PipelineProperties props, DeviceIdentityLookup identities,
                             PropertyDecoder decoder, PresenceHandler presenceHandler, TelemetryWriter writer,
                             LatestValueStore latest, MeterRegistry meters) {
        this.bus = bus;
        this.props = props;
        this.identities = identities;
        this.decoder = decoder;
        this.presenceHandler = presenceHandler;
        this.writer = writer;
        this.latest = latest;
        this.meters = meters;
    }

    public void process(List<Delivery> batch) {
        List<String> presenceAcks = new ArrayList<>();
        List<PendingProperty> properties = new ArrayList<>();

        for (Delivery delivery : batch) {
            if (delivery.deliveryCount() > props.maxDeliveries()) {
                dead(delivery, DeadReason.MAX_DELIVERY_EXCEEDED);
                continue;
            }
            RawMessage message;
            try {
                message = delivery.message();
            } catch (IllegalArgumentException e) {
                dead(delivery, DeadReason.MALFORMED_ENVELOPE);
                continue;
            }
            try {
                switch (message.type()) {
                    case PROPERTY -> properties.add(new PendingProperty(delivery, toPoint(message)));
                    case PRESENCE_CONNECTED, PRESENCE_DISCONNECTED -> {
                        presenceHandler.handle(message);
                        presenceAcks.add(delivery.id());
                        meters.counter("conduit.pipeline.processed", "type", "presence").increment();
                    }
                    case EVENT -> dead(delivery, DeadReason.UNSUPPORTED_TYPE);
                }
            } catch (DecodeException e) {
                log.debug("Dead-lettering {}: {} {}", delivery.id(), e.reason(), e.getMessage());
                dead(delivery, e.reason());
            }
        }

        bus.ack(props.group(), presenceAcks);
        if (!properties.isEmpty()) {
            writeProperties(properties);
        }
    }

    private TelemetryPoint toPoint(RawMessage message) {
        DeviceIdentity identity = identities.find(message.deviceKey())
                .orElseThrow(() -> new DecodeException(DeadReason.UNKNOWN_DEVICE, message.deviceKey()));
        if (!identity.enabled()) {
            throw new DecodeException(DeadReason.DEVICE_DISABLED, message.deviceKey());
        }
        if (!identity.productKey().equals(message.productKey())) {
            throw new DecodeException(DeadReason.PRODUCT_MISMATCH,
                    message.productKey() + " != " + identity.productKey());
        }
        DecodedProperty decoded = decoder.decode(message.payload(), message.receivedAt());
        return new TelemetryPoint(identity.tenantId(), identity.productKey(), identity.deviceKey(),
                decoded.ts(), decoded.params());
    }

    private void writeProperties(List<PendingProperty> items) {
        WriteResult result = writer.write(items.stream().map(PendingProperty::point).toList());
        Set<Integer> rejected = result.rejected().stream()
                .map(WriteResult.Rejected::index).collect(Collectors.toSet());

        List<String> acks = new ArrayList<>();
        for (int i = 0; i < items.size(); i++) {
            PendingProperty item = items.get(i);
            if (rejected.contains(i)) {
                dead(item.delivery(), DeadReason.FIELD_TYPE_CONFLICT);
            } else {
                TelemetryPoint p = item.point();
                latest.update(p.deviceKey(), p.ts(), p.fields());
                acks.add(item.delivery().id());
            }
        }
        bus.ack(props.group(), acks);
        meters.counter("conduit.pipeline.processed", "type", "property").increment(acks.size());
    }

    private void dead(Delivery delivery, DeadReason reason) {
        bus.deadLetter(props.group(), delivery, reason.name());
        meters.counter("conduit.pipeline.dead", "reason", reason.name()).increment();
    }
}
```

```java
package dev.conduit.server.pipeline;

import dev.conduit.common.bus.Delivery;
import dev.conduit.common.bus.MessageBus;
import jakarta.annotation.PostConstruct;
import org.springframework.stereotype.Component;

import java.util.List;

/** 单步执行入口：PipelineWorker 循环调用，测试中直接调用。 */
@Component
public class PipelineRunner {

    private final MessageBus bus;
    private final PipelineProperties props;
    private final PipelineProcessor processor;

    public PipelineRunner(MessageBus bus, PipelineProperties props, PipelineProcessor processor) {
        this.bus = bus;
        this.props = props;
        this.processor = processor;
    }

    @PostConstruct
    void ensureGroup() {
        bus.ensureGroup(props.group());
    }

    public int pollOnce() {
        List<Delivery> batch = bus.read(props.group(), props.consumer(), props.batchSize(), props.block());
        if (!batch.isEmpty()) {
            processor.process(batch);
        }
        return batch.size();
    }

    public int claimOnce() {
        List<Delivery> batch = bus.claimStale(props.group(), props.consumer(), props.claimMinIdle(), props.batchSize());
        if (!batch.isEmpty()) {
            processor.process(batch);
        }
        return batch.size();
    }
}
```

```java
package dev.conduit.server.pipeline;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.context.SmartLifecycle;
import org.springframework.stereotype.Component;

import java.time.Clock;

@Component
@ConditionalOnProperty(prefix = "conduit.pipeline", name = "enabled", havingValue = "true", matchIfMissing = true)
public class PipelineWorker implements SmartLifecycle {

    private static final Logger log = LoggerFactory.getLogger(PipelineWorker.class);
    private static final long INITIAL_BACKOFF_MS = 1_000;
    private static final long MAX_BACKOFF_MS = 30_000;

    private final PipelineRunner runner;
    private final PipelineProperties props;
    private final Clock clock;

    private volatile boolean running;
    private Thread thread;

    public PipelineWorker(PipelineRunner runner, PipelineProperties props, Clock clock) {
        this.runner = runner;
        this.props = props;
        this.clock = clock;
    }

    @Override
    public void start() {
        running = true;
        thread = Thread.ofVirtual().name("pipeline-worker").start(this::loop);
        log.info("Pipeline worker started as consumer {}", props.consumer());
    }

    @Override
    public void stop() {
        running = false;
        if (thread != null) {
            thread.interrupt();
            try {
                thread.join(5_000);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }
    }

    @Override
    public boolean isRunning() {
        return running;
    }

    private void loop() {
        long nextClaimAt = 0;
        long backoff = INITIAL_BACKOFF_MS;
        while (running) {
            try {
                if (clock.millis() >= nextClaimAt) {
                    while (running && runner.claimOnce() > 0) {
                        // 把积压的待处理消息全部认领完
                    }
                    nextClaimAt = clock.millis() + props.claimInterval().toMillis();
                }
                runner.pollOnce();
                backoff = INITIAL_BACKOFF_MS;
            } catch (Exception e) {
                if (!running) {
                    break;
                }
                log.warn("Pipeline iteration failed, retrying in {} ms: {}", backoff, e.getMessage());
                try {
                    Thread.sleep(backoff);
                } catch (InterruptedException ie) {
                    Thread.currentThread().interrupt();
                    break;
                }
                backoff = Math.min(backoff * 2, MAX_BACKOFF_MS);
            }
        }
    }
}
```
