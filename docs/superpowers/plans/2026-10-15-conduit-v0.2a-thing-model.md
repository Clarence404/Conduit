# Conduit V0.2-A Implementation Plan: InfluxDB 3 switch, thing model, pluggable codecs

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace InfluxDB 2.7 with InfluxDB 3 Core 3.12.0, then give every product a thing model (properties) that the pipeline validates per field, with payload decoding selectable per product (`standard-json`, `flat-json`).

**Architecture:** A small JDK-HttpClient `InfluxClient` plus a `LineProtocol` builder replace the InfluxDB 2 Java client; partial writes (`/api/v3/write_lp`, `accept_partial` default) replace the per-point fallback. The thing model lives in PostgreSQL (`product_property`, `product.codec`), is cached per `productKey` in Caffeine (`ThingModelCache`), and the pipeline routes each message through `DecoderRegistry` → `PayloadDecoder` → `FieldValidator` before writing typed values (Long → integer, Double → float). Rejected fields go to the dead-letter stream as `FIELD_REJECTED` records without blocking the accepted fields.

**Tech Stack:** Java 21, Spring Boot 3.5.6, MyBatis-Plus 3.5.12, PostgreSQL 16 + Flyway, Redis 7.4 Streams, InfluxDB 3 Core 3.12.0 (HTTP API v3, SQL), Caffeine (Boot-managed version), Testcontainers, Vue 3 + Element Plus + Vitest.

**Governing spec:** `docs/superpowers/specs/2026-10-15-conduit-v0.2a-thing-model-design.md` (behavior); `docs/superpowers/specs/2026-10-14-conduit-v0.1-design.md` (conventions).

Nothing is silently dead-lettered as a data problem and nothing retries forever.

```java
package dev.conduit.server.telemetry;

import java.util.Map;

/**
 * InfluxDB 行协议构造，只覆盖本项目用到的子集：一个 measurement、若干 tag、若干类型化字段、毫秒时间戳。
 * 字段类型：Long → integer（123i），有限 Double → float（220.0 / 1.0E20），Boolean，String；其余一律拒绝。
 * 行协议无法无歧义表示的名称（含反斜杠或换行、为空）直接拒绝，由调用方按转换失败处理。
 */
public final class LineProtocol {

    private LineProtocol() {
    }

    public static String line(String measurement, Map<String, String> tags, Map<String, Object> fields, long tsMillis) {
        if (fields == null || fields.isEmpty()) {
            throw new IllegalArgumentException("At least one field is required");
        }
        StringBuilder sb = new StringBuilder(64);
        sb.append(escape(measurement, false));
        tags.forEach((key, value) -> sb.append(',').append(escape(key, true)).append('=').append(escape(value, true)));
        char separator = ' ';
        for (Map.Entry<String, Object> field : fields.entrySet()) {
            sb.append(separator).append(escape(field.getKey(), true)).append('=')
                    .append(fieldValue(field.getKey(), field.getValue()));
            separator = ',';
        }
        return sb.append(' ').append(tsMillis).toString();
    }

    static String fieldValue(String name, Object value) {
        return switch (value) {
            case Long l -> l + "i";
            case Double d -> {
                if (!Double.isFinite(d)) {
                    throw new IllegalArgumentException("Non-finite value for field " + name);
                }
                yield Double.toString(d);
            }
            case Boolean b -> b.toString();
            case String s -> '"' + s.replace("\\", "\\\\").replace("\"", "\\\"") + '"';
            case null -> throw new IllegalArgumentException("Null value for field " + name);
            default -> throw new IllegalArgumentException(
                    "Unsupported field type " + value.getClass().getSimpleName() + " for field " + name);
        };
    }

    /** measurement 转义逗号与空格；tag 键值与字段名另外转义等号。 */
    private static String escape(String s, boolean escapeEquals) {
        if (s == null || s.isEmpty()) {
            throw new IllegalArgumentException("Empty measurement, key or tag value");
        }
        StringBuilder sb = new StringBuilder(s.length() + 4);
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c == '\\' || c == '\n' || c == '\r') {
                throw new IllegalArgumentException("Backslash or line break is not supported in '" + s + "'");
            }
            if (c == ',' || c == ' ' || (escapeEquals && c == '=')) {
                sb.append('\\');
            }
            sb.append(c);
        }
        return sb.toString();
    }
}
```

```java
package dev.conduit.server.telemetry;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.io.IOException;
import java.io.InterruptedIOException;
import java.io.UncheckedIOException;
import java.net.URI;
import java.net.URLEncoder;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;

/**
 * InfluxDB 3 Core HTTP API（v3）的最小客户端。IO 失败抛 UncheckedIOException；非 2xx 抛 InfluxHttpException，
 * 唯一例外是写入时的部分行被拒（400 + partial write JSON），此时返回被拒的行。
 */
public class InfluxClient implements AutoCloseable {

    /** 被拒的一行；lineNumber 从 1 开始，对应提交顺序。 */
    public record LineError(int lineNumber, String message) {
    }

    static final String PARTIAL_WRITE = "partial write of line protocol occurred";
    private static final Duration CONNECT_TIMEOUT = Duration.ofSeconds(5);
    private static final Duration REQUEST_TIMEOUT = Duration.ofSeconds(10);
    private static final TypeReference<List<Map<String, Object>>> ROWS = new TypeReference<>() {
    };

    private final String baseUrl;
    private final String authorization;
    private final HttpClient http;
    private final ObjectMapper mapper = new ObjectMapper();

    public InfluxClient(URI baseUrl, String token) {
        String url = baseUrl.toString();
        this.baseUrl = url.endsWith("/") ? url.substring(0, url.length() - 1) : url;
        this.authorization = "Bearer " + token;
        this.http = HttpClient.newBuilder().connectTimeout(CONNECT_TIMEOUT).build();
    }

    /** 写入行协议（accept_partial 默认开启：合格行照常写入）。全部成功返回空列表。 */
    public List<LineError> writeLines(String db, List<String> lines) {
        HttpResponse<String> response = send(HttpRequest.newBuilder(
                        uri("/api/v3/write_lp", "db", db, "precision", "millisecond"))
                .header("Content-Type", "text/plain; charset=utf-8")
                .POST(HttpRequest.BodyPublishers.ofString(String.join("\n", lines), StandardCharsets.UTF_8)));
        int status = response.statusCode();
        if (status == 204 || status == 200) {
            return List.of();
        }
        if (status == 400) {
            List<LineError> errors = partialWriteErrors(response.body());
            if (errors != null) {
                return errors;
            }
        }
        throw new InfluxHttpException(status, response.body());
    }

    public List<Map<String, Object>> querySql(String db, String sql) {
        HttpResponse<String> response = send(HttpRequest.newBuilder(
                uri("/api/v3/query_sql", "db", db, "q", sql, "format", "json")).GET());
        if (response.statusCode() != 200) {
            throw new InfluxHttpException(response.statusCode(), response.body());
        }
        String body = response.body();
        if (body == null || body.isBlank()) {
            return List.of();
        }
        try {
            return mapper.readValue(body, ROWS);
        } catch (JsonProcessingException e) {
            throw new IllegalStateException("Unexpected InfluxDB query response: " + e.getOriginalMessage(), e);
        }
    }

    /** 保留期只能在建库时设置；库已存在（409）视为成功。 */
    public void createDatabaseIfMissing(String db, String retention) {
        String body;
        try {
            body = mapper.writeValueAsString(Map.of("db", db, "retention_period", retention));
        } catch (JsonProcessingException e) {
            throw new IllegalStateException(e);
        }
        HttpResponse<String> response = send(HttpRequest.newBuilder(uri("/api/v3/configure/database"))
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString(body, StandardCharsets.UTF_8)));
        int status = response.statusCode();
        if (status / 100 == 2 || status == 409) {
            return;
        }
        throw new InfluxHttpException(status, response.body());
    }

    @Override
    public void close() {
        http.close();
    }

    private List<LineError> partialWriteErrors(String body) {
        try {
            JsonNode root = mapper.readTree(body);
            if (root == null || !PARTIAL_WRITE.equals(root.path("error").asText()) || !root.path("data").isArray()) {
                return null;
            }
            List<LineError> errors = new ArrayList<>();
            for (JsonNode item : root.get("data")) {
                errors.add(new LineError(item.path("line_number").asInt(-1), item.path("error_message").asText("")));
            }
            return errors;
        } catch (JsonProcessingException e) {
            return null;
        }
    }

    private URI uri(String path, String... query) {
        StringBuilder sb = new StringBuilder(baseUrl).append(path);
        for (int i = 0; i < query.length; i += 2) {
            sb.append(i == 0 ? '?' : '&').append(query[i]).append('=')
                    .append(URLEncoder.encode(query[i + 1], StandardCharsets.UTF_8));
        }
        return URI.create(sb.toString());
    }

    private HttpResponse<String> send(HttpRequest.Builder request) {
        try {
            return http.send(request.timeout(REQUEST_TIMEOUT).header("Authorization", authorization).build(),
                    HttpResponse.BodyHandlers.ofString(StandardCharsets.UTF_8));
        } catch (IOException e) {
            throw new UncheckedIOException(e);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new UncheckedIOException(new InterruptedIOException("Interrupted while calling InfluxDB"));
        }
    }
}
```

```java
package dev.conduit.server.telemetry;

import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

@Component
public class TelemetryWriter {

    static final String MEASUREMENT = "property";

    private final InfluxClient client;
    private final InfluxProperties props;

    public TelemetryWriter(InfluxClient client, InfluxProperties props) {
        this.client = client;
        this.props = props;
    }

    /**
     * 一次请求写入整批。无法转换的点（不支持的类型、null 等）在写入前拒绝；InfluxDB 逐行拒绝的点（通常是字段类型冲突）
     * 按 line_number 映射回原始下标；其余行已写入。下游不可用抛 TelemetryUnavailableException（可原地重试）；
     * 非逐行拒绝的 400 说明请求本身有问题，抛 IllegalStateException（不可原地重试，见实施计划 Task 1 的决定）。
     */
    public WriteResult write(List<TelemetryPoint> points) {
        if (points.isEmpty()) {
            return WriteResult.ok();
        }
        List<WriteResult.Rejected> rejected = new ArrayList<>();
        List<Integer> indexes = new ArrayList<>();
        List<String> lines = new ArrayList<>();
        for (int i = 0; i < points.size(); i++) {
            try {
                lines.add(toLine(points.get(i)));
                indexes.add(i);
            } catch (RuntimeException e) {
                rejected.add(new WriteResult.Rejected(i, String.valueOf(e.getMessage())));
            }
        }
        if (lines.isEmpty()) {
            return new WriteResult(rejected);
        }
        for (InfluxClient.LineError error : send(lines)) {
            int line = error.lineNumber() - 1;
            if (line < 0 || line >= indexes.size()) {
                throw new IllegalStateException("InfluxDB reported rejected line " + error.lineNumber()
                        + " outside the batch of " + lines.size());
            }
            rejected.add(new WriteResult.Rejected(indexes.get(line), error.message()));
        }
        rejected.sort(Comparator.comparingInt(WriteResult.Rejected::index));
        return new WriteResult(rejected);
    }

    private List<InfluxClient.LineError> send(List<String> lines) {
        try {
            return client.writeLines(props.database(), lines);
        } catch (InfluxHttpException e) {
            if (e.status() == 400) {
                throw new IllegalStateException("InfluxDB rejected the write request itself: " + e.getMessage(), e);
            }
            throw new TelemetryUnavailableException(e);
        } catch (RuntimeException e) {
            throw new TelemetryUnavailableException(e);
        }
    }

    static String toLine(TelemetryPoint p) {
        Map<String, String> tags = new LinkedHashMap<>();
        tags.put("tenant_id", Long.toString(p.tenantId()));
        tags.put("product_key", p.productKey());
        tags.put("device_key", p.deviceKey());
        return LineProtocol.line(MEASUREMENT, tags, p.fields(), p.ts());
    }
}
```

```java
import java.util.List;

/** enum_items（jsonb）与 List<EnumItem> 互转。以 Types.OTHER 绑定字符串，PostgreSQL 才会按 jsonb 接收。 */
public class EnumItemsTypeHandler extends BaseTypeHandler<List<EnumItem>> {

    private static final ObjectMapper MAPPER = new ObjectMapper();
    private static final TypeReference<List<EnumItem>> TYPE = new TypeReference<>() {
    };

    @Override
    public void setNonNullParameter(PreparedStatement ps, int i, List<EnumItem> parameter, JdbcType jdbcType)
            throws SQLException {
        try {
            ps.setObject(i, MAPPER.writeValueAsString(parameter), Types.OTHER);
        } catch (JsonProcessingException e) {
            throw new SQLException("Cannot serialize enum items", e);
        }
    }

    @Override
    public List<EnumItem> getNullableResult(ResultSet rs, String columnName) throws SQLException {
        return parse(rs.getString(columnName));
    }

    @Override
    public List<EnumItem> getNullableResult(ResultSet rs, int columnIndex) throws SQLException {
        return parse(rs.getString(columnIndex));
    }

    @Override
    public List<EnumItem> getNullableResult(CallableStatement cs, int columnIndex) throws SQLException {
        return parse(cs.getString(columnIndex));
    }

    private static List<EnumItem> parse(String json) throws SQLException {
        if (json == null) {
            return null;
        }
        try {
            return MAPPER.readValue(json, TYPE);
        } catch (JsonProcessingException e) {
            throw new SQLException("Cannot parse enum items", e);
        }
    }
}
```

```java
package dev.conduit.server.pipeline.codec;

import com.fasterxml.jackson.databind.JsonNode;
import dev.conduit.server.pipeline.DeadReason;
import dev.conduit.server.pipeline.DecodeException;
import org.springframework.stereotype.Component;

import java.time.Clock;
import java.time.Duration;

/** 所有解码器共用的时间戳规则：毫秒整数，范围 [now − 30 天, now + 5 分钟]，缺省取接收时间。 */
@Component
public class TimestampPolicy {

    static final Duration MAX_PAST = Duration.ofDays(30);
    static final Duration MAX_FUTURE = Duration.ofMinutes(5);

    private final Clock clock;

    public TimestampPolicy(Clock clock) {
        this.clock = clock;
    }

    public long resolve(JsonNode ts, long receivedAt) {
        if (ts == null || ts.isNull()) {
            return receivedAt;
        }
        if (!ts.isIntegralNumber() || !ts.canConvertToLong()) {
            throw new DecodeException(DeadReason.TS_OUT_OF_RANGE, "ts must be integer milliseconds");
        }
        return check(ts.longValue());
    }

    public long check(long ts) {
        long now = clock.millis();
        if (ts < now - MAX_PAST.toMillis() || ts > now + MAX_FUTURE.toMillis()) {
            throw new DecodeException(DeadReason.TS_OUT_OF_RANGE, "ts " + ts + " outside allowed window");
        }
        return ts;
    }
}
```

```java
package dev.conduit.server.pipeline.codec;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.DeserializationFeature;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.ObjectReader;
import dev.conduit.server.pipeline.DeadReason;
import dev.conduit.server.pipeline.DecodeException;

import java.io.IOException;

/** JSON 解码器共用的解析与取值规则。 */
final class JsonPayloads {

    private JsonPayloads() {
    }

    /** 拒绝重复键与尾随字符。 */
    static ObjectReader strictReader(ObjectMapper mapper) {
        return mapper.reader().with(DeserializationFeature.FAIL_ON_READING_DUP_TREE_KEY,
                DeserializationFeature.FAIL_ON_TRAILING_TOKENS);
    }

    static JsonNode readObject(ObjectReader reader, byte[] payload) {
        JsonNode root;
        try {
            root = reader.readTree(payload);
        } catch (JsonProcessingException e) {
            // 不使用 getMessage()：它会附带 [Source: ...] 原始报文片段，进而泄漏到死信与日志
            throw new DecodeException(DeadReason.MALFORMED_JSON, e.getOriginalMessage());
        } catch (IOException e) {
            throw new DecodeException(DeadReason.MALFORMED_JSON, "unreadable payload");
        }
        if (root == null || !root.isObject()) {
            throw new DecodeException(DeadReason.MALFORMED_JSON, "payload is not a JSON object");
        }
        return root;
    }

    /** 测点名不能为空、不能以 __ 开头（平台保留）。 */
    static String checkName(String name) {
        if (name.isBlank() || name.startsWith("__")) {
            throw new DecodeException(DeadReason.INVALID_PARAMS, "invalid field name '" + name + "'");
        }
        return name;
    }

    /** 保留原始类型：Long / BigInteger / Double（可能非有限，由校验决定）/ String / Boolean。 */
    static Object rawValue(String name, JsonNode v) {
        if (v.isIntegralNumber()) {
            return v.canConvertToLong() ? (Object) v.longValue() : v.bigIntegerValue();
        }
        if (v.isFloatingPointNumber()) {
            return v.doubleValue();
        }
        if (v.isTextual()) {
            return v.textValue();
        }
        if (v.isBoolean()) {
            return v.booleanValue();
        }
        throw new DecodeException(DeadReason.INVALID_PARAMS, "unsupported value type for '" + name + "'");
    }
}
```

```java
package dev.conduit.server.pipeline.codec;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.ObjectReader;
import dev.conduit.server.pipeline.DeadReason;
import dev.conduit.server.pipeline.DecodeException;
import org.springframework.stereotype.Component;

import java.util.LinkedHashMap;
import java.util.Map;

/** V0.1 标准格式：{"id":..., "ts":毫秒, "params":{测点: 数值|字符串|布尔}}。 */
@Component
public class StandardJsonDecoder implements PayloadDecoder {

    public static final String NAME = "standard-json";

    private final ObjectReader reader;
    private final TimestampPolicy timestamps;

    public StandardJsonDecoder(ObjectMapper objectMapper, TimestampPolicy timestamps) {
        this.reader = JsonPayloads.strictReader(objectMapper);
        this.timestamps = timestamps;
    }

    @Override
    public String name() {
        return NAME;
    }

    @Override
    public String displayName() {
        return "标准 JSON";
    }

    @Override
    public DecodedPayload decode(DecodeContext ctx, byte[] payload) {
        JsonNode root = JsonPayloads.readObject(reader, payload);
        long ts = timestamps.resolve(root.get("ts"), ctx.receivedAt());
        JsonNode params = root.get("params");
        if (params == null || !params.isObject() || params.isEmpty()) {
            throw new DecodeException(DeadReason.INVALID_PARAMS, "params must be a non-empty object");
        }
        Map<String, Object> values = new LinkedHashMap<>();
        for (Map.Entry<String, JsonNode> field : params.properties()) {
            String name = JsonPayloads.checkName(field.getKey());
            values.put(name, JsonPayloads.rawValue(name, field.getValue()));
        }
        return new DecodedPayload(ts, values);
    }
}
```

```java
import java.util.Map;

/** 扁平格式：{"ts":毫秒, "Ua":220.1, ...}。顶层 ts 为时间戳，id 忽略，其余键均为测点，至少一个。 */
@Component
public class FlatJsonDecoder implements PayloadDecoder {

    public static final String NAME = "flat-json";

    private final ObjectReader reader;
    private final TimestampPolicy timestamps;

    public FlatJsonDecoder(ObjectMapper objectMapper, TimestampPolicy timestamps) {
        this.reader = JsonPayloads.strictReader(objectMapper);
        this.timestamps = timestamps;
    }

    @Override
    public String name() {
        return NAME;
    }

    @Override
    public String displayName() {
        return "扁平 JSON";
    }

    @Override
    public DecodedPayload decode(DecodeContext ctx, byte[] payload) {
        JsonNode root = JsonPayloads.readObject(reader, payload);
        long ts = timestamps.resolve(root.get("ts"), ctx.receivedAt());
        Map<String, Object> values = new LinkedHashMap<>();
        for (Map.Entry<String, JsonNode> field : root.properties()) {
            String name = field.getKey();
            if ("ts".equals(name) || "id".equals(name)) {
                continue;
            }
            JsonPayloads.checkName(name);
            values.put(name, JsonPayloads.rawValue(name, field.getValue()));
        }
        if (values.isEmpty()) {
            throw new DecodeException(DeadReason.INVALID_PARAMS, "payload has no fields besides ts and id");
        }
        return new DecodedPayload(ts, values);
    }
}
```

```java
package dev.conduit.server.pipeline.codec;

import org.springframework.stereotype.Component;

import java.util.Collections;
import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.TreeMap;

/** 收集所有 PayloadDecoder Bean 并按 name 索引；重名时启动失败。 */
@Component
public class DecoderRegistry {

    private final Map<String, PayloadDecoder> byName;

    public DecoderRegistry(List<PayloadDecoder> decoders) {
        Map<String, PayloadDecoder> map = new TreeMap<>();
        for (PayloadDecoder decoder : decoders) {
            PayloadDecoder previous = map.putIfAbsent(decoder.name(), decoder);
            if (previous != null) {
                throw new IllegalStateException("Duplicate payload decoder name '" + decoder.name() + "': "
                        + previous.getClass().getName() + " and " + decoder.getClass().getName());
            }
        }
        this.byName = Collections.unmodifiableMap(map);
    }

    public Optional<PayloadDecoder> find(String name) {
        return name == null ? Optional.empty() : Optional.ofNullable(byName.get(name));
    }

    public boolean exists(String name) {
        return find(name).isPresent();
    }

    public List<PayloadDecoder> all() {
        return List.copyOf(byName.values());
    }
}
```

```java
package dev.conduit.server.pipeline;

import dev.conduit.server.thingmodel.PropertySpec;

import java.math.BigInteger;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

/**
 * 按物模型逐测点校验并转换为写入类型（V0.2-A spec 4.2）。
 * 没有物模型时与 V0.1 一致：数值一律转 Double，文本、布尔原样，非有限数值使整条消息 INVALID_PARAMS。
 */
public final class FieldValidator {

    static final int MAX_TEXT_LENGTH = 1024;
    private static final double LONG_MIN = -0x1p63;
    private static final double LONG_MAX_EXCLUSIVE = 0x1p63;

    private FieldValidator() {
    }

    public static FieldValidation validate(Map<String, PropertySpec> model, Map<String, Object> values) {
        return model.isEmpty() ? withoutModel(values) : withModel(model, values);
    }

    private static FieldValidation withoutModel(Map<String, Object> values) {
        Map<String, Object> accepted = new LinkedHashMap<>();
        values.forEach((name, value) -> accepted.put(name, switch (value) {
            case Number n -> {
                double d = n.doubleValue();
                // 超出 double 范围的字面量会变成 ±Infinity，写入 InfluxDB 会失败
                if (!Double.isFinite(d)) {
                    throw new DecodeException(DeadReason.INVALID_PARAMS, "non-finite number for '" + name + "'");
                }
                yield d;
            }
            case String s -> s;
            case Boolean b -> b;
            case null, default ->
                    throw new DecodeException(DeadReason.INVALID_PARAMS, "unsupported value type for '" + name + "'");
        }));
        return new FieldValidation(accepted, List.of());
    }

    private static FieldValidation withModel(Map<String, PropertySpec> model, Map<String, Object> values) {
        Map<String, Object> accepted = new LinkedHashMap<>();
        List<RejectedField> rejected = new ArrayList<>();
        values.forEach((name, value) -> {
            PropertySpec spec = model.get(name);
            if (spec == null) {
                rejected.add(new RejectedField(name, value, RejectReason.UNDEFINED));
                return;
            }
            Outcome outcome = check(spec, value);
            if (outcome.reason() == null) {
                accepted.put(name, outcome.value());
            } else {
                rejected.add(new RejectedField(name, value, outcome.reason()));
            }
        });
        return new FieldValidation(accepted, rejected);
    }

    private record Outcome(Object value, RejectReason reason) {
        static Outcome accept(Object value) {
            return new Outcome(value, null);
        }

        static Outcome reject(RejectReason reason) {
            return new Outcome(null, reason);
        }
    }

    private static Outcome check(PropertySpec spec, Object value) {
        return switch (spec.dataType()) {
            case DOUBLE -> {
                Double d = finiteDouble(value);
                if (d == null) {
                    yield Outcome.reject(RejectReason.TYPE_MISMATCH);
                }
                yield inRange(spec, d) ? Outcome.accept(d) : Outcome.reject(RejectReason.OUT_OF_RANGE);
            }
            case INT -> {
                Long l = integer(value);
                if (l == null) {
                    yield Outcome.reject(RejectReason.TYPE_MISMATCH);
                }
                yield inRange(spec, l.doubleValue()) ? Outcome.accept(l) : Outcome.reject(RejectReason.OUT_OF_RANGE);
            }
            case BOOL -> value instanceof Boolean ? Outcome.accept(value) : Outcome.reject(RejectReason.TYPE_MISMATCH);
            case TEXT -> {
                if (!(value instanceof String s)) {
                    yield Outcome.reject(RejectReason.TYPE_MISMATCH);
                }
                yield s.length() <= MAX_TEXT_LENGTH ? Outcome.accept(s) : Outcome.reject(RejectReason.TOO_LONG);
            }
            case ENUM -> {
                Long l = integer(value);
                if (l == null) {
                    yield Outcome.reject(RejectReason.TYPE_MISMATCH);
                }
                yield spec.enumLabels().containsKey(l) ? Outcome.accept(l) : Outcome.reject(RejectReason.ENUM_VALUE);
            }
        };
    }

    /** Long、BigInteger 或 Double 中的有限数值；其他返回 null。 */
    static Double finiteDouble(Object value) {
        if (value instanceof Long || value instanceof BigInteger || value instanceof Double) {
            double d = ((Number) value).doubleValue();
            return Double.isFinite(d) ? d : null;
        }
        return null;
    }

    /** 整数判断：Long、long 范围内的 BigInteger、值为整数且在 long 范围内的 Double（如 220.0）。 */
    static Long integer(Object value) {
        return switch (value) {
            case Long l -> l;
            case BigInteger b -> b.bitLength() < 64 ? b.longValue() : null;
            case Double d -> Double.isFinite(d) && d == Math.rint(d) && d >= LONG_MIN && d < LONG_MAX_EXCLUSIVE
                    ? d.longValue() : null;
            case null, default -> null;
        };
    }

    private static boolean inRange(PropertySpec spec, double v) {
        return (spec.minValue() == null || v >= spec.minValue()) && (spec.maxValue() == null || v <= spec.maxValue());
    }
}
```

```java
package dev.conduit.server.thingmodel;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.github.benmanes.caffeine.cache.Cache;
import com.github.benmanes.caffeine.cache.Caffeine;
import com.github.benmanes.caffeine.cache.Ticker;
import dev.conduit.server.product.Product;
import dev.conduit.server.product.ProductMapper;
import dev.conduit.server.tenant.TenantContext;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.time.Duration;
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Optional;

/**
 * 按 productKey 缓存 ProductProfile：写入后 60 秒过期，最多 10 000 项。跨租户查询。
 * 产品不存在时不缓存（由设备身份检查兜底）。修改属性、导入、修改 codec、删除产品时由服务层主动 evict；
 * 多实例下其他实例最多延迟 60 秒。读库异常原样抛出，流水线按可恢复错误原地重试。
 */
@Component
public class ThingModelCache {

    static final Duration TTL = Duration.ofSeconds(60);
    static final long MAX_SIZE = 10_000;

    private final ProductMapper products;
    private final ProductPropertyMapper properties;
    private final Cache<String, ProductProfile> cache;

    @Autowired
    public ThingModelCache(ProductMapper products, ProductPropertyMapper properties) {
        this(products, properties, Ticker.systemTicker());
    }

    ThingModelCache(ProductMapper products, ProductPropertyMapper properties, Ticker ticker) {
        this.products = products;
        this.properties = properties;
        this.cache = Caffeine.newBuilder().expireAfterWrite(TTL).maximumSize(MAX_SIZE).ticker(ticker).build();
    }

    public Optional<ProductProfile> find(String productKey) {
        return Optional.ofNullable(cache.get(productKey, this::load));
    }

    public void evict(String productKey) {
        cache.invalidate(productKey);
    }

    private ProductProfile load(String productKey) {
        Product product = products.findByKeyAnyTenant(productKey);
        if (product == null) {
            return null;
        }
        List<ProductProperty> rows = TenantContext.callAs(product.getTenantId(), () -> properties.selectList(
                new LambdaQueryWrapper<ProductProperty>()
                        .eq(ProductProperty::getProductId, product.getId())
                        .orderByAsc(ProductProperty::getId)));
        Map<String, PropertySpec> specs = new LinkedHashMap<>();
        rows.forEach(row -> specs.put(row.getIdentifier(), PropertySpec.of(row)));
        return new ProductProfile(product.getCodec(), Collections.unmodifiableMap(specs));
    }
}
```

  - `ProductDtos.SaveProductRequest(String name, String description, String codec)`; `ProductDtos.ProductView(String productKey, String name, String description, String codec, int propertyCount, OffsetDateTime createdAt)`.

**Decisions recorded here:**
- The export item has **9** fields (`identifier, name, dataType, unit, minValue, maxValue, decimals, enumItems, description`); spec §5 says "8 个字段", which counts the range as one. Nothing else (id, timestamps) is exported.
- `PUT /products/{pk}` without `codec` keeps the current codec (only `POST` defaults to `standard-json`), so the list page's edit dialog cannot silently reset a flat-json product.
- Identifiers `time`, `tenant_id`, `product_key`, `device_key` are rejected (they collide with InfluxDB 3 columns of table `property` and every write would be rejected as a column type conflict).
- `PUT /properties/{identifier}`: the path identifier wins; a body `identifier` is ignored (renaming is not supported). A body `dataType` different from the stored one → 409 「数据类型不可修改」, checked before the other rules.
- Create and import lock the product row (`select … for update`) so the 200-property limit holds under concurrency. Cache eviction happens after commit.

```java
package dev.conduit.server.thingmodel;

import dev.conduit.server.thingmodel.ThingModelDtos.EnumItemRequest;
import dev.conduit.server.thingmodel.ThingModelDtos.PropertyRequest;

import java.math.BigInteger;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;
import java.util.regex.Pattern;

/** 属性校验（新增、修改、导入共用，V0.2-A spec 5「属性校验」）。 */
final class PropertyRules {

    static final Pattern IDENTIFIER = Pattern.compile("^[A-Za-z][A-Za-z0-9_]{0,63}$");
    /** 与 InfluxDB 3 表 property 的时间列、tag 列同名，写入必然被拒。 */
    static final Set<String> RESERVED_IDENTIFIERS = Set.of("time", "tenant_id", "product_key", "device_key");
    static final int MAX_PROPERTIES = 200;
    static final int MAX_ENUM_ITEMS = 64;
    static final int MAX_NAME = 64;
    static final int MAX_UNIT = 16;
    static final int MAX_DESCRIPTION = 256;

    private PropertyRules() {
    }

    static DataType parseType(String value) {
        if (value == null) {
            return null;
        }
        try {
            return DataType.valueOf(value);
        } catch (IllegalArgumentException e) {
            return null;
        }
    }

    /** 返回全部错误，为空表示合法。 */
    static List<String> check(PropertyRequest r) {
        List<String> errors = new ArrayList<>();
        String id = r.identifier();
        if (id == null || !IDENTIFIER.matcher(id).matches()) {
            errors.add("标识须以字母开头，只含字母、数字、下划线，最长 64 个字符");
        } else if (RESERVED_IDENTIFIERS.contains(id)) {
            errors.add("标识 " + id + " 为保留字");
        }
        if (r.name() == null || r.name().isBlank()) {
            errors.add("名称不能为空");
        } else if (r.name().length() > MAX_NAME) {
            errors.add("名称最长 64 个字符");
        }
        DataType type = parseType(r.dataType());
        if (type == null) {
            errors.add("数据类型必须是 DOUBLE、INT、BOOL、TEXT、ENUM 之一");
        }
        if (r.unit() != null && r.unit().length() > MAX_UNIT) {
            errors.add("单位最长 16 个字符");
        }
        if (r.description() != null && r.description().length() > MAX_DESCRIPTION) {
            errors.add("说明最长 256 个字符");
        }
        if (r.decimals() != null && (r.decimals() < 0 || r.decimals() > 6)) {
            errors.add("小数位数须在 0～6 之间");
        }
        Double min = r.minValue();
        Double max = r.maxValue();
        if ((min != null || max != null) && type != null && type != DataType.DOUBLE && type != DataType.INT) {
            errors.add("只有数值类型可以设置范围");
        }
        if ((min != null && !Double.isFinite(min)) || (max != null && !Double.isFinite(max))) {
            errors.add("范围必须是有限数值");
        } else if (min != null && max != null && min > max) {
            errors.add("最小值不能大于最大值");
        }
        List<EnumItemRequest> items = r.enumItems() == null ? List.of() : r.enumItems();
        if (type == DataType.ENUM) {
            errors.addAll(checkEnumItems(items));
        } else if (type != null && !items.isEmpty()) {
            errors.add("只有枚举类型可以设置枚举项");
        }
        return errors;
    }

    private static List<String> checkEnumItems(List<EnumItemRequest> items) {
        List<String> errors = new ArrayList<>();
        if (items.isEmpty()) {
            errors.add("枚举类型至少需要一个枚举项");
            return errors;
        }
        if (items.size() > MAX_ENUM_ITEMS) {
            errors.add("枚举项最多 64 个");
        }
        Set<Long> seen = new HashSet<>();
        for (EnumItemRequest item : items) {
            if (item == null) {
                errors.add("枚举项不能为空");
                continue;
            }
            Long value = integerValue(item.value());
            if (value == null) {
                errors.add("枚举值必须是整数：" + item.value());
            } else if (!seen.add(value)) {
                errors.add("枚举值重复：" + value);
            }
            if (item.label() == null || item.label().isBlank()) {
                errors.add("枚举项名称不能为空");
            }
        }
        return errors;
    }

    static Long integerValue(Number n) {
        return switch (n) {
            case Long l -> l;
            case Integer i -> i.longValue();
            case Short s -> s.longValue();
            case Byte b -> b.longValue();
            case BigInteger b -> b.bitLength() < 64 ? b.longValue() : null;
            case null, default -> null;
        };
    }

    /** 把合法请求写入实体（调用前 check 必须为空）。identifier、dataType 由调用方处理。 */
    static void apply(PropertyRequest r, ProductProperty target) {
        target.setName(r.name().trim());
        target.setUnit(blankToNull(r.unit()));
        target.setMinValue(r.minValue());
        target.setMaxValue(r.maxValue());
        target.setDecimals(r.decimals());
        target.setEnumItems(parseType(r.dataType()) == DataType.ENUM
                ? r.enumItems().stream().map(i -> new EnumItem(integerValue(i.value()), i.label().trim())).toList()
                : null);
        target.setDescription(blankToNull(r.description()));
    }

    private static String blankToNull(String s) {
        return s == null || s.isBlank() ? null : s.trim();
    }
}
```

```java
package dev.conduit.server.thingmodel;

import dev.conduit.server.thingmodel.ThingModelDtos.PropertyRequest;
import dev.conduit.server.web.ItemErrorsException.ItemError;

import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

/** 导入前校验整个文件，收集全部错误；任一错误则整体不生效。 */
final class ThingModelImport {

    private ThingModelImport() {
    }

    static List<ItemError> validate(List<PropertyRequest> items, Map<String, DataType> existingTypes, ImportMode mode) {
        List<ItemError> errors = new ArrayList<>();
        Set<String> identifiers = new HashSet<>();
        for (int i = 0; i < items.size(); i++) {
            PropertyRequest item = items.get(i);
            if (item == null) {
                errors.add(new ItemError(null, "第 " + (i + 1) + " 项为空"));
                continue;
            }
            String id = item.identifier();
            PropertyRules.check(item).forEach(message -> errors.add(new ItemError(id, message)));
            if (id != null && !identifiers.add(id)) {
                errors.add(new ItemError(id, "标识重复"));
            }
            DataType requested = PropertyRules.parseType(item.dataType());
            DataType current = id == null ? null : existingTypes.get(id);
            if (current != null && requested != null && current != requested) {
                errors.add(new ItemError(id, "数据类型不可修改（现为 " + current + "）"));
            }
        }
        int total;
        if (mode == ImportMode.REPLACE) {
            total = identifiers.size();
        } else {
            Set<String> union = new HashSet<>(existingTypes.keySet());
            union.addAll(identifiers);
            total = union.size();
        }
        if (total > PropertyRules.MAX_PROPERTIES) {
            errors.add(new ItemError(null, "导入后属性数为 " + total + "，超过上限 " + PropertyRules.MAX_PROPERTIES));
        }
        return errors;
    }
}
```

```java
/** 属性增删改查与物模型导入导出。所有修改在提交后清除该产品的物模型缓存。 */
@Service
public class ThingModelService {

    private final ProductService products;
    private final ProductMapper productMapper;
    private final ProductPropertyMapper properties;
    private final ThingModelCache cache;
    private final TransactionTemplate tx;
    private final Clock clock;

    public ThingModelService(ProductService products, ProductMapper productMapper, ProductPropertyMapper properties,
                             ThingModelCache cache, TransactionTemplate tx, Clock clock) {
        this.products = products;
        this.productMapper = productMapper;
        this.properties = properties;
        this.cache = cache;
        this.tx = tx;
        this.clock = clock;
    }

    public List<PropertyView> list(String productKey) {
        return rows(products.require(productKey).getId()).stream().map(PropertyView::of).toList();
    }

    public PropertyView create(String productKey, PropertyRequest request) {
        Product product = products.require(productKey);
        requireValid(request);
        ProductProperty created = tx.execute(status -> {
            productMapper.lockById(product.getId());
            if (findRow(product.getId(), request.identifier()) != null) {
                throw ApiException.conflict("属性标识已存在: " + request.identifier());
            }
            if (properties.countByProduct(product.getId()) >= PropertyRules.MAX_PROPERTIES) {
                throw ApiException.conflict("每个产品最多 " + PropertyRules.MAX_PROPERTIES + " 个属性");
            }
            ProductProperty row = newRow(product.getId(), request);
            properties.insert(row);
            return properties.selectById(row.getId());
        });
        cache.evict(productKey);
        return PropertyView.of(created);
    }

    public PropertyView update(String productKey, String identifier, PropertyRequest request) {
        Product product = products.require(productKey);
        ProductProperty row = requireRow(product.getId(), identifier);
        PropertyRequest effective = request.withIdentifier(identifier);
        DataType requested = PropertyRules.parseType(effective.dataType());
        if (requested != null && requested != row.getDataType()) {
            throw ApiException.conflict("数据类型不可修改");
        }
        requireValid(effective);
        PropertyRules.apply(effective, row);
        row.setUpdatedAt(OffsetDateTime.now(clock));
        properties.updateById(row);
        cache.evict(productKey);
        return PropertyView.of(properties.selectById(row.getId()));
    }

    public void delete(String productKey, String identifier) {
        Product product = products.require(productKey);
        properties.deleteById(requireRow(product.getId(), identifier).getId());
        cache.evict(productKey);
    }

    public ThingModelDocument export(String productKey) {
        return new ThingModelDocument(rows(products.require(productKey).getId()).stream()
                .map(PropertyRequest::of).toList());
    }

    public List<PropertyView> importModel(String productKey, ThingModelDocument document, ImportMode mode) {
        Product product = products.require(productKey);
        if (document == null || document.properties() == null) {
            throw ApiException.badRequest("文件缺少 properties 数组");
        }
        List<ProductProperty> result = tx.execute(status -> {
            productMapper.lockById(product.getId());
            Map<String, ProductProperty> existing = new LinkedHashMap<>();
            rows(product.getId()).forEach(r -> existing.put(r.getIdentifier(), r));
            Map<String, DataType> existingTypes = new LinkedHashMap<>();
            existing.forEach((id, r) -> existingTypes.put(id, r.getDataType()));
            List<ItemError> errors = ThingModelImport.validate(document.properties(), existingTypes, mode);
            if (!errors.isEmpty()) {
                throw new ItemErrorsException("物模型导入失败，共 " + errors.size() + " 处错误，未做任何修改", errors);
            }
            OffsetDateTime now = OffsetDateTime.now(clock);
            Set<String> inFile = new HashSet<>();
            for (PropertyRequest item : document.properties()) {
                inFile.add(item.identifier());
                ProductProperty row = existing.get(item.identifier());
                if (row == null) {
                    properties.insert(newRow(product.getId(), item));
                } else {
                    PropertyRules.apply(item, row);
                    row.setUpdatedAt(now);
                    properties.updateById(row);
                }
            }
            if (mode == ImportMode.REPLACE) {
                existing.values().stream()
                        .filter(r -> !inFile.contains(r.getIdentifier()))
                        .forEach(r -> properties.deleteById(r.getId()));
            }
            return rows(product.getId());
        });
        cache.evict(productKey);
        return result.stream().map(PropertyView::of).toList();
    }

    private static ProductProperty newRow(long productId, PropertyRequest request) {
        ProductProperty row = new ProductProperty();
        row.setProductId(productId);
        row.setIdentifier(request.identifier());
        row.setDataType(DataType.valueOf(request.dataType()));
        PropertyRules.apply(request, row);
        return row;
    }

    private static void requireValid(PropertyRequest request) {
        List<String> errors = PropertyRules.check(request);
        if (!errors.isEmpty()) {
            throw ApiException.badRequest(String.join("；", errors));
        }
    }

    private List<ProductProperty> rows(long productId) {
        return properties.selectList(new LambdaQueryWrapper<ProductProperty>()
                .eq(ProductProperty::getProductId, productId).orderByAsc(ProductProperty::getId));
    }

    private ProductProperty findRow(long productId, String identifier) {
        return properties.selectOne(new LambdaQueryWrapper<ProductProperty>()
                .eq(ProductProperty::getProductId, productId).eq(ProductProperty::getIdentifier, identifier));
    }

    private ProductProperty requireRow(long productId, String identifier) {
        ProductProperty row = findRow(productId, identifier);
        if (row == null) {
            throw ApiException.notFound("属性不存在: " + identifier);
        }
        return row;
    }
```

```java
package dev.conduit.server.pipeline;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

/** 死信记录中的 rejected 字段：[{"field":"Ua","value":"abc","reason":"TYPE_MISMATCH"}]。 */
final class RejectedFields {

    static final String KEY = "rejected";
    static final int MAX_VALUE_LENGTH = 64;
    private static final ObjectMapper MAPPER = new ObjectMapper();

    private RejectedFields() {
    }

    static Map<String, String> details(List<RejectedField> rejected) {
        List<Map<String, String>> items = rejected.stream().map(r -> {
            Map<String, String> item = new LinkedHashMap<>();
            item.put("field", r.field());
            item.put("value", truncate(String.valueOf(r.value())));
            item.put("reason", r.reason().name());
            return item;
        }).toList();
        try {
            return Map.of(KEY, MAPPER.writeValueAsString(items));
        } catch (JsonProcessingException e) {
            throw new IllegalStateException(e);
        }
    }

    static String truncate(String value) {
        if (value.codePointCount(0, value.length()) <= MAX_VALUE_LENGTH) {
            return value;
        }
        return value.substring(0, value.offsetByCodePoints(0, MAX_VALUE_LENGTH));
    }
}
```

```java
                      Duration initialBackoff, Duration maxBackoff) {

    public void process(List<Delivery> batch) {
        List<String> presenceAcks = new ArrayList<>();
        List<PendingProperty> properties = new ArrayList<>();

        for (Delivery delivery : batch) {
            if (delivery.deliveryCount() > props.maxDeliveries()) {
                dead(delivery, DeadReason.MAX_DELIVERY_EXCEEDED);
                continue;
            }
            RawMessage message;
            try {
                message = delivery.message();
            } catch (IllegalArgumentException e) {
                dead(delivery, DeadReason.MALFORMED_ENVELOPE);
                continue;
            }
            try {
                switch (message.type()) {
                    case PROPERTY -> properties.add(retry("lookup", () -> toPending(delivery, message)));
                    case PRESENCE_CONNECTED, PRESENCE_DISCONNECTED -> {
                        retry("presence", () -> {
                            presenceHandler.handle(message);
                            return null;
                        });
                        presenceAcks.add(delivery.id());
                        meters.counter("conduit.pipeline.processed", "type", "presence").increment();
                    }
                    case EVENT -> dead(delivery, DeadReason.UNSUPPORTED_TYPE);
                }
            } catch (DecodeException e) {
                log.debug("Dead-lettering {}: {} {}", delivery.id(), e.reason(), e.getMessage());
                dead(delivery, e.reason(), e.details());
            } catch (ProcessingAbortedException e) {
                throw e;
            } catch (RuntimeException e) {
                log.error("Unexpected failure processing delivery {}", delivery.id(), e);
                dead(delivery, DeadReason.PROCESSING_ERROR);
            }
        }

        ack(presenceAcks);
        if (!properties.isEmpty()) {
            writeProperties(properties);
        }

            throw new DecodeException(DeadReason.PRODUCT_MISMATCH,
                    message.productKey() + " != " + identity.productKey());
        }
        ProductProfile profile = thingModels.find(identity.productKey())
                .orElseThrow(() -> new DecodeException(DeadReason.UNKNOWN_DEVICE,
                        "product not found: " + identity.productKey()));
        PayloadDecoder decoder = decoders.find(profile.codec())
                .orElseThrow(() -> new DecodeException(DeadReason.UNKNOWN_CODEC, String.valueOf(profile.codec())));
        DecodedPayload decoded = decoder.decode(
                new DecodeContext(identity.productKey(), identity.deviceKey(), message.receivedAt()),
                message.payload());
        FieldValidation validation = FieldValidator.validate(profile.properties(), decoded.values());
        validation.rejected().forEach(r ->
                meters.counter("conduit.pipeline.rejected_fields", "reason", r.reason().name()).increment());
        if (validation.accepted().isEmpty()) {
            throw new DecodeException(DeadReason.FIELD_REJECTED, "all fields rejected",
                    RejectedFields.details(validation.rejected()));
        }
        TelemetryPoint point = new TelemetryPoint(identity.tenantId(), identity.productKey(), identity.deviceKey(),
                decoded.ts(), validation.accepted());
        return new PendingProperty(delivery, point, validation.rejected());
    }

    private void writeProperties(List<PendingProperty> items) {
        List<TelemetryPoint> points = items.stream().map(PendingProperty::point).toList();
        WriteResult result = retry("influx write", () -> writer.write(points));
        Set<Integer> rejected = result.rejected().stream()
                .map(WriteResult.Rejected::index).collect(Collectors.toSet());

        List<String> acks = new ArrayList<>();
        for (int i = 0; i < items.size(); i++) {
            PendingProperty item = items.get(i);
            if (rejected.contains(i)) {
                dead(item.delivery(), DeadReason.FIELD_TYPE_CONFLICT);
                continue;
            }
            TelemetryPoint p = item.point();
            try {
                retry("latest update", () -> {
                    latest.update(p.deviceKey(), p.ts(), p.fields());
                    return null;
                });
                if (!item.rejected().isEmpty()) {
                    Map<String, String> details = RejectedFields.details(item.rejected());
                    // 先留痕再 ACK：两步之间崩溃只会产生重复死信记录，不会丢失
                    retry("dead record", () -> {
                        bus.recordDead(item.delivery(), DeadReason.FIELD_REJECTED.name(), details);
                        return null;
                    });
                }
                acks.add(item.delivery().id());
            } catch (ProcessingAbortedException e) {
                throw e;
            } catch (RuntimeException e) {
                log.error("Unexpected failure finishing delivery {}", item.delivery().id(), e);
                dead(item.delivery(), DeadReason.PROCESSING_ERROR);
            }
        }
        ack(acks);
        meters.counter("conduit.pipeline.processed", "type", "property").increment(acks.size());
    }

    private void ack(List<String> ids) {
        if (ids.isEmpty()) {
            return;
        }
        retry("ack", () -> {
            bus.ack(props.group(), ids);
            return null;
        });
    }

    /** 仅暂时性故障值得原地重试；其余（语法错误、完整性冲突、WRONGTYPE 等）重试也不会成功。 */
    private static boolean isRecoverable(Throwable e) {
        return e instanceof TelemetryUnavailableException
                || e instanceof DataAccessResourceFailureException
                || e instanceof TransientDataAccessException
                || e instanceof RecoverableDataAccessException;
    }

    /** 下游暂时不可用时原地指数退避重试，不重新读取，因此不消耗投递次数。 */
    private <T> T retry(String step, Supplier<T> action) {
        long delay = initialBackoff.toMillis();
        for (int attempt = 1; ; attempt++) {
            try {
                return action.get();
            } catch (RuntimeException e) {
                if (!isRecoverable(e)) {
                    throw e;
                }
                log.warn("{} unavailable (attempt {}), retrying in {} ms: {}", step, attempt, delay, e.getMessage());
                try {
                    Thread.sleep(delay);
                } catch (InterruptedException ie) {
                    Thread.currentThread().interrupt();
                    throw new ProcessingAbortedException("Interrupted while retrying " + step);
                }
                delay = Math.min(delay * 2, maxBackoff.toMillis());
            }
        }
    }

    private void dead(Delivery delivery, DeadReason reason) {
        dead(delivery, reason, Map.of());
    }

    private void dead(Delivery delivery, DeadReason reason, Map<String, String> details) {
        retry("dead letter", () -> {
            if (details.isEmpty()) {
                bus.deadLetter(props.group(), delivery, reason.name());
            } else {
                bus.recordDead(delivery, reason.name(), details);
                bus.ack(props.group(), List.of(delivery.id()));
            }
            return null;
        });
        meters.counter("conduit.pipeline.dead", "reason", reason.name()).increment();
    }
}
```

```java
import dev.conduit.server.device.DeviceService;
import dev.conduit.server.thingmodel.DataType;

import dev.conduit.server.thingmodel.PropertySpec;

import org.springframework.web.bind.annotation.PathVariable;

import java.util.Map;

@RestController
public class TelemetryController {

    /** name/unit/decimals 在物模型中无定义时为 null；enumLabel 仅枚举且值已定义时有值。 */
    public record PointView(String identifier, String name, Object value, long ts, String unit, Integer decimals,
                            String enumLabel) {
    }

    public record LatestView(String deviceKey, List<PointView> values) {
    }

    private final DeviceService devices;
    private final LatestValueStore latest;
    private final ThingModelCache thingModels;

    public TelemetryController(DeviceService devices, LatestValueStore latest, ThingModelCache thingModels) {
        this.devices = devices;
        this.latest = latest;
        this.thingModels = thingModels;
    }

    @GetMapping("/api/v1/devices/{deviceKey}/latest")
    public LatestView latest(@PathVariable String deviceKey) {
        String productKey = devices.get(deviceKey).productKey();
        Map<String, PropertySpec> model = thingModels.find(productKey)
                .map(ProductProfile::properties).orElse(Map.of());
        List<PointView> values = latest.get(deviceKey).entrySet().stream()
                .map(e -> view(e.getKey(), e.getValue(), model.get(e.getKey())))
                .toList();
        return new LatestView(deviceKey, values);
    }

    static PointView view(String identifier, LatestValue v, PropertySpec spec) {
        if (spec == null) {
            return new PointView(identifier, null, v.value(), v.ts(), null, null, null);
        }
        return new PointView(identifier, spec.name(), v.value(), v.ts(), spec.unit(), spec.decimals(),
                enumLabel(spec, v.value()));
    }

    static String enumLabel(PropertySpec spec, Object value) {
        if (spec.dataType() != DataType.ENUM || !(value instanceof Number n)) {
            return null;
        }
        double d = n.doubleValue();
        if (d != Math.rint(d)) {
            return null;
        }
        return spec.enumLabels().get(n.longValue());
    }
```

```java
package dev.conduit.simulator;

import java.util.Locale;
import java.util.Map;

/** 上报格式，与平台解码器一一对应。 */
public enum PayloadFormat {

    STANDARD("standard-json") {
        @Override
        public byte[] encode(String id, long ts, Map<String, Object> params) {
            return PayloadFactory.property(id, ts, params);
        }
    },
    FLAT("flat-json") {
        @Override
        public byte[] encode(String id, long ts, Map<String, Object> params) {
            return PayloadFactory.flat(id, ts, params);
        }
    };

    private final String codec;

    PayloadFormat(String codec) {
        this.codec = codec;
    }

    public String codec() {
        return codec;
    }

    public abstract byte[] encode(String id, long ts, Map<String, Object> params);

    public static PayloadFormat parse(String value) {
        return switch (value.trim().toLowerCase(Locale.ROOT)) {
            case "standard" -> STANDARD;
            case "flat" -> FLAT;
            default -> throw new IllegalStateException("SIM_PAYLOAD_FORMAT must be standard or flat, got: " + value);
        };
    }
}
```

```ts
export interface ProblemFieldError {
  identifier: string | null
  message: string
}

/** ProblemDetail 中的 errors 列表（如物模型导入的逐项错误）；没有时返回空数组。 */
export function problemErrors(error: unknown): ProblemFieldError[] {
  if (!isAxiosError(error)) {
    return []
  }
  const errors = (error.response?.data as { errors?: unknown } | undefined)?.errors
  if (!Array.isArray(errors)) {
    return []
  }
  return errors
    .filter((e): e is { identifier?: unknown; message?: unknown } => typeof e === 'object' && e !== null)
    .map((e) => ({
      identifier: typeof e.identifier === 'string' ? e.identifier : null,
      message: typeof e.message === 'string' ? e.message : '',
    }))
}
```

```ts
export interface FieldVisibility {
  range: boolean
  decimals: boolean
  enumItems: boolean
  intHint: boolean

export function fieldVisibility(type: DataType): FieldVisibility {
  const numeric = type === 'DOUBLE' || type === 'INT'
  return {
    range: numeric,
    decimals: numeric,
    enumItems: type === 'ENUM',
    intHint: type === 'INT' || type === 'ENUM',

/** 弹窗表单：数值字段用字符串，便于区分「未填写」与 0 */
export interface PropertyForm {
  identifier: string
  name: string
  dataType: DataType
  unit: string
  minValue: string
  maxValue: string
  decimals: string
  enumItems: { value: string; label: string }[]
  description: string
}

export function emptyForm(): PropertyForm {
  return {
    identifier: '',
    name: '',
    dataType: 'DOUBLE',
    unit: '',
    minValue: '',
    maxValue: '',
    decimals: '',
    enumItems: [],
    description: '',
  }

const text = (n: number | null) => (n === null ? '' : String(n))

export function formFromProperty(p: PropertyView): PropertyForm {
  return {
    identifier: p.identifier,
    name: p.name,
    dataType: p.dataType,
    unit: p.unit ?? '',
    minValue: text(p.minValue),
    maxValue: text(p.maxValue),
    decimals: text(p.decimals),
    enumItems: (p.enumItems ?? []).map((i) => ({ value: String(i.value), label: i.label })),
    description: p.description ?? '',
  }
}

function blankToNull(s: string): string | null {
  const t = s.trim()
  return t ? t : null
}

function optionalNumber(s: string, label: string): number | null {
  const t = s.trim()
  if (!t) {
    return null
  }
  const n = Number(t)
  if (!Number.isFinite(n)) {
    throw new Error(`${label}必须是数字`)
  }
  return n
}

function optionalInteger(s: string, label: string): number | null {
  const n = optionalNumber(s, label)
  if (n !== null && !Number.isInteger(n)) {
    throw new Error(`${label}必须是整数`)
  }
  return n
}

/** 表单转请求：隐藏的字段一律为 null；格式错误抛出 Error（message 可直接提示用户） */
export function toPropertyRequest(f: PropertyForm): PropertyRequest {
  const v = fieldVisibility(f.dataType)
  return {
    identifier: f.identifier.trim(),
    name: f.name.trim(),
    dataType: f.dataType,
    unit: blankToNull(f.unit),
    minValue: v.range ? optionalNumber(f.minValue, '最小值') : null,
    maxValue: v.range ? optionalNumber(f.maxValue, '最大值') : null,
    decimals: v.decimals ? optionalInteger(f.decimals, '小数位数') : null,
    enumItems: v.enumItems
      ? f.enumItems.map((item, i) => {
          const value = optionalInteger(item.value, `第 ${i + 1} 个枚举值`)
          if (value === null) {
            throw new Error(`第 ${i + 1} 个枚举值不能为空`)
          }
          return { value, label: item.label.trim() }
        })
      : null,
    description: blankToNull(f.description),
  }
}

export function formatRange(min: number | null, max: number | null): string {
  if (min !== null && max !== null) {
    return `${min} ~ ${max}`
  }
  if (min !== null) {
    return `≥ ${min}`
  }
  if (max !== null) {
    return `≤ ${max}`
  }
  return '—'
}

export function codecLabel(codecs: CodecView[], name: string): string {
  return codecs.find((c) => c.name === name)?.displayName ?? name
}
```
