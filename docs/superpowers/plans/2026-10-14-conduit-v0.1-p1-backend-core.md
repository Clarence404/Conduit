# Conduit V0.1 · P1 后端主干（common + connector）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建好 Maven 多模块骨架，实现统一消息模型与 Redis Streams 总线，并让 connector 把 EMQX 上的设备消息与上下线事件可靠地投递到总线。

**Architecture:** `conduit-common` 提供 `RawMessage`、`MessageBus` 接口与 `RedisStreamMessageBus` 实现；`conduit-connector` 通过 `ProtocolAdapter` 接口挂载 `MqttProtocolAdapter`，用 HiveMQ 客户端以持久会话 + 共享订阅 + 手动 ACK 订阅 EMQX，XADD 成功后才回 PUBACK。

**Tech Stack:** Java 21、Spring Boot 3.5.6、Spring Data Redis（Lettuce）、HiveMQ MQTT Client 1.3.7、JUnit 5、AssertJ、Awaitility、Testcontainers（Spring Boot 管理版本）

本计划是 V0.1 三份计划中的第 1 份：P1 后端主干 → P2 server → P3 模拟器/前端/部署/端到端。设计依据：`docs/superpowers/specs/2026-10-14-conduit-v0.1-design.md`。

## Global Constraints

- Java 21；Spring Boot 3.5.6（父 POM）；groupId `dev.conduit`；版本 `0.1.0-SNAPSHOT`；根包 `dev.conduit`。
- 单元测试类名 `*Test`（surefire），需要容器的集成测试类名 `*IT`（failsafe）；`mvn verify` 两者都跑。
- 运行集成测试需要本机 Docker Desktop 已启动。
- 镜像版本固定：`redis:7.4.2`、`emqx/emqx:5.8.6`；不得使用 `latest`。
- Redis key：上行总线 `conduit:stream:up`（`MAXLEN ~ 1000000`），死信 `conduit:stream:dead`（`MAXLEN ~ 100000`）。
- 设备 topic：`conduit/{productKey}/{deviceKey}/up/property`、`conduit/{productKey}/{deviceKey}/up/event`；上下线系统主题 `$SYS/brokers/{node}/clients/{clientid}/connected|disconnected`。
- connector 只搬运原始字节，不解析业务 payload。
- 提交信息使用 Conventional Commits，不加任何 AI 署名。
- 所有命令在仓库根目录 仓库根目录 下用 Git Bash 执行。

```java
package dev.conduit.common.message;

import java.util.Map;
import java.util.Objects;

/**
 * 协议无关的原始消息。connector 只负责组装它，业务解析在 server 完成。
 *
 * @param productKey 产品 Key；上下线事件拿不到时为 null
 * @param deviceKey  设备 Key（等于 MQTT clientId）
 * @param payload    原始字节，不做任何解析
 * @param receivedAt 平台接收时间（毫秒）
 * @param attributes 协议相关的附加信息，例如原始 topic
 */
public record RawMessage(String productKey, String deviceKey, MessageType type, byte[] payload,
                         long receivedAt, Map<String, String> attributes) {

    public RawMessage {
        Objects.requireNonNull(deviceKey, "deviceKey");
        Objects.requireNonNull(type, "type");
        payload = payload == null ? new byte[0] : payload;
        attributes = attributes == null ? Map.of() : Map.copyOf(attributes);
    }
}
```

`backend/conduit-common/src/main/java/dev/conduit/common/message/RawMessageCodec.java`：

```java
package dev.conduit.common.message;

import java.util.Base64;
import java.util.LinkedHashMap;
import java.util.Map;

/** RawMessage 与 Redis Stream 字段之间的转换。payload 用 Base64 存储以支持二进制协议。 */
public final class RawMessageCodec {

    static final String PRODUCT_KEY = "pk";
    static final String DEVICE_KEY = "dk";
    static final String TYPE = "type";
    static final String PAYLOAD = "payload";
    static final String RECEIVED_AT = "receivedAt";
    static final String ATTR_PREFIX = "attr.";

    private RawMessageCodec() {
    }

    public static Map<String, String> toFields(RawMessage message) {
        Map<String, String> fields = new LinkedHashMap<>();
        if (message.productKey() != null) {
            fields.put(PRODUCT_KEY, message.productKey());
        }
        fields.put(DEVICE_KEY, message.deviceKey());
        fields.put(TYPE, message.type().name());
        fields.put(PAYLOAD, Base64.getEncoder().encodeToString(message.payload()));
        fields.put(RECEIVED_AT, Long.toString(message.receivedAt()));
        message.attributes().forEach((k, v) -> fields.put(ATTR_PREFIX + k, v));
        return fields;
    }

    public static RawMessage fromFields(Map<String, String> fields) {
        String deviceKey = require(fields, DEVICE_KEY);
        MessageType type = MessageType.valueOf(require(fields, TYPE));
        byte[] payload = Base64.getDecoder().decode(require(fields, PAYLOAD));
        long receivedAt = Long.parseLong(require(fields, RECEIVED_AT));

        Map<String, String> attributes = new LinkedHashMap<>();
        fields.forEach((k, v) -> {
            if (k.startsWith(ATTR_PREFIX)) {
                attributes.put(k.substring(ATTR_PREFIX.length()), v);
            }
        });
        return new RawMessage(fields.get(PRODUCT_KEY), deviceKey, type, payload, receivedAt, attributes);
    }

    private static String require(Map<String, String> fields, String name) {
        String value = fields.get(name);
        if (value == null) {
            throw new IllegalArgumentException("missing field: " + name);
        }
        return value;
    }
}
```

`MessageType.valueOf` 与 `Long.parseLong` 在非法输入时分别抛 `IllegalArgumentException` 与其子类 `NumberFormatException`，满足接口约定。

```java
package dev.conduit.common.bus;

import dev.conduit.common.message.RawMessage;

import java.time.Duration;
import java.util.List;

/**
 * 接入与处理之间的消息总线。V0.1 由 Redis Streams 实现，后续可替换为 Kafka / RabbitMQ。
 * 语义为至少一次投递：消费者处理成功后必须 ack，否则消息会被重新认领。
 */
public interface MessageBus {

    String publish(RawMessage message);

    void ensureGroup(String group);

    List<Delivery> read(String group, String consumer, int count, Duration block);

    List<Delivery> claimStale(String group, String consumer, Duration minIdle, int count);

    void ack(String group, List<String> ids);

    void deadLetter(String group, Delivery delivery, String reason);
```

```java
package dev.conduit.common.bus;

import dev.conduit.common.message.RawMessage;
import dev.conduit.common.message.RawMessageCodec;
import org.springframework.data.domain.Range;
import org.springframework.data.redis.RedisSystemException;
import org.springframework.data.redis.connection.RedisStreamCommands.XAddOptions;
import org.springframework.data.redis.connection.stream.Consumer;
import org.springframework.data.redis.connection.stream.MapRecord;
import org.springframework.data.redis.connection.stream.PendingMessage;
import org.springframework.data.redis.connection.stream.PendingMessages;
import org.springframework.data.redis.connection.stream.ReadOffset;
import org.springframework.data.redis.connection.stream.RecordId;
import org.springframework.data.redis.connection.stream.StreamOffset;
import org.springframework.data.redis.connection.stream.StreamReadOptions;
import org.springframework.data.redis.connection.stream.StreamRecords;
import org.springframework.data.redis.core.RedisCallback;
import org.springframework.data.redis.core.StringRedisTemplate;

import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

public class RedisStreamMessageBus implements MessageBus {

    private final StringRedisTemplate redis;
    private final MessageBusProperties props;

    public RedisStreamMessageBus(StringRedisTemplate redis, MessageBusProperties props) {
        this.redis = redis;
        this.props = props;
    }

    @Override
    public String publish(RawMessage message) {
        return xadd(props.streamKey(), RawMessageCodec.toFields(message), props.maxLen());
    }

    @Override
    public void ensureGroup(String group) {
        byte[] key = bytes(props.streamKey());
        try {
            redis.execute((RedisCallback<String>) c ->
                    c.streamCommands().xGroupCreate(key, group, ReadOffset.from("0-0"), true));
        } catch (RedisSystemException e) {
            String msg = String.valueOf(e.getMostSpecificCause().getMessage());
            if (!msg.contains("BUSYGROUP")) {
                throw e;
            }
        }
    }

    @Override
    public List<Delivery> read(String group, String consumer, int count, Duration block) {
        @SuppressWarnings("unchecked")
        List<MapRecord<String, Object, Object>> records = redis.opsForStream().read(
                Consumer.from(group, consumer),
                StreamReadOptions.empty().count(count).block(block),
                StreamOffset.create(props.streamKey(), ReadOffset.lastConsumed()));
        if (records == null) {
            return List.of();
        }
        return records.stream()
                .map(r -> new Delivery(r.getId().getValue(), toStringMap(r.getValue()), 1))
                .toList();
    }

    @Override
    public List<Delivery> claimStale(String group, String consumer, Duration minIdle, int count) {
        PendingMessages pending = redis.opsForStream()
                .pending(props.streamKey(), group, Range.unbounded(), count);
        List<PendingMessage> stale = pending.stream()
                .filter(p -> p.getElapsedTimeSinceLastDelivery().compareTo(minIdle) >= 0)
                .toList();
        if (stale.isEmpty()) {
            return List.of();
        }
        Map<String, Long> previousCounts = stale.stream()
                .collect(Collectors.toMap(PendingMessage::getIdAsString, PendingMessage::getTotalDeliveryCount));

        List<MapRecord<String, Object, Object>> claimed = redis.opsForStream().claim(
                props.streamKey(), group, consumer, minIdle,
                stale.stream().map(PendingMessage::getId).toArray(RecordId[]::new));

        return claimed.stream()
                .map(r -> {
                    String id = r.getId().getValue();
                    return new Delivery(id, toStringMap(r.getValue()), previousCounts.getOrDefault(id, 0L) + 1);
                })
                .toList();
    }

    @Override
    public void ack(String group, List<String> ids) {
        if (ids.isEmpty()) {
            return;
        }
        redis.opsForStream().acknowledge(props.streamKey(), group, ids.toArray(String[]::new));
    }

    @Override
    public void deadLetter(String group, Delivery delivery, String reason) {
        Map<String, String> fields = new LinkedHashMap<>(delivery.fields());
        fields.put("reason", reason);
        fields.put("originalId", delivery.id());
        fields.put("deliveryCount", Long.toString(delivery.deliveryCount()));
        fields.put("failedAt", Long.toString(System.currentTimeMillis()));
        xadd(props.deadKey(), fields, props.deadMaxLen());
        ack(group, List.of(delivery.id()));
    }

    private String xadd(String key, Map<String, String> fields, long maxLen) {
        Map<byte[], byte[]> raw = new LinkedHashMap<>();
        fields.forEach((k, v) -> raw.put(bytes(k), bytes(v)));
        RecordId id = redis.execute((RedisCallback<RecordId>) c -> c.streamCommands().xAdd(
                StreamRecords.rawBytes(raw).withStreamKey(bytes(key)),
                XAddOptions.maxlen(maxLen).approximateTrimming(true)));
        if (id == null) {
            throw new IllegalStateException("XADD returned no id for " + key);
        }
        return id.getValue();
    }

    private static Map<String, String> toStringMap(Map<Object, Object> value) {
        Map<String, String> result = new LinkedHashMap<>();
        value.forEach((k, v) -> result.put(String.valueOf(k), String.valueOf(v)));
        return result;
    }

    private static byte[] bytes(String s) {
        return s.getBytes(StandardCharsets.UTF_8);
    }
}
```

```java
package dev.conduit.connector.mqtt;

import dev.conduit.common.message.MessageType;

import java.util.Optional;

public final class TopicParser {

    private TopicParser() {
    }

    public static Optional<ParsedTopic> parse(String topic) {
        return topic.startsWith("$SYS/") ? parseSystemTopic(topic) : parseDeviceTopic(topic);
    }

    /** conduit/{productKey}/{deviceKey}/up/{property|event} */
    private static Optional<ParsedTopic> parseDeviceTopic(String topic) {
        String[] p = topic.split("/", -1);
        if (p.length != 5 || !"conduit".equals(p[0]) || !"up".equals(p[3])
                || p[1].isEmpty() || p[2].isEmpty()) {
            return Optional.empty();
        }
        MessageType type = switch (p[4]) {
            case "property" -> MessageType.PROPERTY;
            case "event" -> MessageType.EVENT;
            default -> null;
        };
        return Optional.ofNullable(type).map(t -> new ParsedTopic(p[1], p[2], t));
    }

    /** $SYS/brokers/{node}/clients/{clientid}/{connected|disconnected} */
    private static Optional<ParsedTopic> parseSystemTopic(String topic) {
        String[] p = topic.split("/", -1);
        if (p.length != 6 || !"brokers".equals(p[1]) || !"clients".equals(p[3]) || p[4].isEmpty()) {
            return Optional.empty();
        }
        MessageType type = switch (p[5]) {
            case "connected" -> MessageType.PRESENCE_CONNECTED;
            case "disconnected" -> MessageType.PRESENCE_DISCONNECTED;
            default -> null;
        };
        return Optional.ofNullable(type).map(t -> new ParsedTopic(null, p[4], t));
    }
}
```

```java
package dev.conduit.connector.mqtt;

import com.hivemq.client.mqtt.MqttClient;
import com.hivemq.client.mqtt.MqttGlobalPublishFilter;
import com.hivemq.client.mqtt.datatypes.MqttQos;
import com.hivemq.client.mqtt.mqtt5.Mqtt5AsyncClient;
import com.hivemq.client.mqtt.mqtt5.Mqtt5ClientBuilder;
import com.hivemq.client.mqtt.mqtt5.message.publish.Mqtt5Publish;
import dev.conduit.common.bus.MessageBus;
import dev.conduit.common.message.MessageType;
import dev.conduit.common.message.RawMessage;
import dev.conduit.connector.adapter.ProtocolAdapter;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.nio.charset.StandardCharsets;
import java.time.Clock;
import java.util.Map;
import java.util.Optional;
import java.util.concurrent.Executors;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;

/**
 * 以持久会话 + 共享订阅接收设备消息，XADD 成功后才回 PUBACK（手动 ACK），保证至少一次投递。
 */
public class MqttProtocolAdapter implements ProtocolAdapter {

    private static final Logger log = LoggerFactory.getLogger(MqttProtocolAdapter.class);
    private static final long MAX_RETRY_DELAY_MS = 30_000;

    private final MqttConnectorProperties props;
    private final MessageBus bus;
    private final Clock clock;
    private final ScheduledExecutorService retryExecutor = Executors.newSingleThreadScheduledExecutor(
            Thread.ofPlatform().name("mqtt-bus-retry").daemon().factory());

    private Mqtt5AsyncClient client;

    public MqttProtocolAdapter(MqttConnectorProperties props, MessageBus bus, Clock clock) {
        this.props = props;
        this.bus = bus;
        this.clock = clock;
    }

    @Override
    public String name() {
        return "mqtt";
    }

    @Override
    public void start() {
        Mqtt5ClientBuilder builder = MqttClient.builder().useMqttVersion5()
                .identifier(props.clientId())
                .serverHost(props.host())
                .serverPort(props.port())
                .automaticReconnectWithDefaultConfig();
        if (props.username() != null && !props.username().isBlank()) {
            builder = builder.simpleAuth()
                    .username(props.username())
                    .password(String.valueOf(props.password()).getBytes(StandardCharsets.UTF_8))
                    .applySimpleAuth();
        }
        client = builder.buildAsync();

        // 先注册全局回调再连接：持久会话重连后 broker 会直接推送积压消息
        client.publishes(MqttGlobalPublishFilter.SUBSCRIBED, this::onPublish, true);

        client.connectWith()
                .cleanStart(false)
                .sessionExpiryInterval(props.sessionExpirySeconds())
                .send().join();

        client.subscribeWith()
                .addSubscription().topicFilter("$share/" + props.shareGroup() + "/conduit/+/+/up/#")
                .qos(MqttQos.AT_LEAST_ONCE).applySubscription()
                // 上下线事件按事件时间幂等处理，多实例重复投递无害，因此不用共享订阅
                .addSubscription().topicFilter("$SYS/brokers/+/clients/+/connected")
                .qos(MqttQos.AT_LEAST_ONCE).applySubscription()
                .addSubscription().topicFilter("$SYS/brokers/+/clients/+/disconnected")
                .qos(MqttQos.AT_LEAST_ONCE).applySubscription()
                .send().join();

        log.info("MQTT adapter connected to {}:{} as {}", props.host(), props.port(), props.clientId());
    }

    @Override
    public void stop() {
        retryExecutor.shutdownNow();
        if (client != null && client.getState().isConnected()) {
            client.disconnect().join();
        }
    }

    void onPublish(Mqtt5Publish publish) {
        String topic = publish.getTopic().toString();
        Optional<ParsedTopic> parsed = TopicParser.parse(topic);
        if (parsed.isEmpty()) {
            log.warn("Dropping unroutable topic {}", topic);
            publish.acknowledge();
            return;
        }
        ParsedTopic t = parsed.get();
        if (isPresence(t.type()) && t.deviceKey().startsWith(props.ownClientPrefix())) {
            publish.acknowledge();
            return;
        }
        RawMessage message = new RawMessage(t.productKey(), t.deviceKey(), t.type(),
                publish.getPayloadAsBytes(), clock.millis(), Map.of("topic", topic));
        publishWithRetry(message, publish, 0);
    }

    private void publishWithRetry(RawMessage message, Mqtt5Publish publish, int attempt) {
        try {
            bus.publish(message);
            publish.acknowledge();
        } catch (RuntimeException e) {
            long delay = Math.min(MAX_RETRY_DELAY_MS, 1000L << Math.min(attempt, 5));
            log.warn("Bus publish failed for device {} (attempt {}), retrying in {} ms: {}",
                    message.deviceKey(), attempt + 1, delay, e.getMessage());
            retryExecutor.schedule(() -> publishWithRetry(message, publish, attempt + 1),
                    delay, TimeUnit.MILLISECONDS);
        }
    }

    private static boolean isPresence(MessageType type) {
        return type == MessageType.PRESENCE_CONNECTED || type == MessageType.PRESENCE_DISCONNECTED;
    }
}
```

```java
package dev.conduit.connector.adapter;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.context.SmartLifecycle;
import org.springframework.stereotype.Component;

import java.util.List;

@Component
public class AdapterLifecycle implements SmartLifecycle {

    private static final Logger log = LoggerFactory.getLogger(AdapterLifecycle.class);

    private final List<ProtocolAdapter> adapters;
    private volatile boolean running;

    public AdapterLifecycle(List<ProtocolAdapter> adapters) {
        this.adapters = adapters;
    }

    @Override
    public void start() {
        adapters.forEach(a -> {
            log.info("Starting protocol adapter {}", a.name());
            a.start();
        });
        running = true;
    }

    @Override
    public void stop() {
        adapters.forEach(ProtocolAdapter::stop);
        running = false;
    }

    @Override
    public boolean isRunning() {
        return running;
    }
}
```
